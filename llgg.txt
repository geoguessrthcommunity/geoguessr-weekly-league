  //   Visionneuse plein écran du logo

  //   Classement provisoire des équipes (règles de la Ligue, seuil de 10 %)
  //   Item "équipe LLGG" dans le switch des résultats
  //   Tuile flottante "Prochaine journée" (accueil, décompte, notification)
  //   Lien "La Ligue" dans le menu header
  //   Injection dans le DOM (player-nick_wrapper + activity-card_groupActor)
  //   Onglet "équipe LLGG" dans les classements (rounds, récap du défi du jour)
  //   Récap du défi du jour : coéquipiers amis, rang global, détail
  //   Sélection de division sur le site de l'organisation (2e @match)
  //   Observer global + démarrage
  // =================================================================

  // Passer à true pour retrouver les logs de diagnostic détaillés utiles
  // pendant le développement (données chargées, non-correspondances...).
  // Laissé à false pour un usage normal — évite de spammer la console
  // des joueurs qui n'en ont pas besoin.
  const DEBUG = false;
  function logDebug(...args) {
    if (DEBUG) console.debug("[LLGG]", ...args);
  }

  // Le script est aussi déclaré sur laliguegeoguessr.fr (voir @match),
  // mais UNIQUEMENT pour y sélectionner une division à l'arrivée depuis
  // un de nos liens — voir demarrerSiteLigue en bas de fichier. Cette
  // constante sert à court-circuiter tout le reste là-bas, y compris les
  // écouteurs enregistrés au niveau racine.
  const SUR_SITE_LIGUE = location.hostname === "www.laliguegeoguessr.fr";

  // ---------------------------------------------------------------
  // File d'attente réseau (requêtes de fond vers geoguessr.com)
  // ---------------------------------------------------------------
  // Deux fonctionnalités interrogent les pages de profil en tâche de
  // fond : les drapeaux (fiche d'équipe) et la vérification des pseudos
  // (anti-faux-positif). Sans limite, une liste d'amis bien remplie
  // déclenchait des dizaines de requêtes SIMULTANÉES vers GeoGuessr dès
  // l'affichage — inoffensif pour un usage perso, mais c'est typiquement
  // ce qui fait bloquer un script diffusé à toute une communauté. On
  // plafonne donc le nombre de requêtes en vol ; le reste patiente.
  // 8 : une fiche d'équipe complète (8 membres) part donc d'un seul coup,
  // sans attente perceptible, tout en gardant une vraie protection contre
  // les rafales de plusieurs dizaines de requêtes (liste d'amis fournie).
  const MAX_REQUETES_SIMULTANEES = 8;
  let requetesEnVol = 0;
  // Deux niveaux : "prioritaire" pour ce que l'utilisateur attend à
  // l'écran (drapeaux d'une fiche ouverte), "normal" pour le travail de
  // fond invisible (vérification des pseudos). Sans ça, ouvrir une fiche
  // juste après l'arrivée sur une liste d'amis coinçait les drapeaux
  // derrière des dizaines de vérifications déjà en file.
  const fileRequetes = { prioritaire: [], normal: [] };

  function traiterFileRequetes() {
    while (requetesEnVol < MAX_REQUETES_SIMULTANEES) {
      const tache = fileRequetes.prioritaire.shift() || fileRequetes.normal.shift();
      if (!tache) return;
      requetesEnVol++;
      tache()
        .catch(() => {}) // l'appelant gère déjà ses propres erreurs
        .finally(() => {
          requetesEnVol--;
          traiterFileRequetes();
        });
    }
  }

  // Renvoie une promesse résolue avec le résultat de `travail`, mais
  // n'exécute celui-ci que lorsqu'un créneau se libère.
  function mettreEnFile(travail, prioritaire = false) {
    return new Promise((resolve, reject) => {
      const file = prioritaire ? fileRequetes.prioritaire : fileRequetes.normal;
      file.push(() => travail().then(resolve, reject));
      traiterFileRequetes();
    });
  }

  // ---------------------------------------------------------------
  // Réglages & cache
  // ---------------------------------------------------------------
  const CLE_CACHE_DONNEES = "laligue_donnees_cache";
  const CLE_ACTIF = "laligue_actif";
  const CLE_MASQUER_CLUB_NATIF = "laligue_masquer_club_natif";
  const DUREE_CACHE_MS = 60 * 60 * 1000; // 1h
  // Incrémenté à chaque changement de la structure retournée par
  // fetchDonneesLigueDepuisSource() — évite d'avoir à vider le cache
  // manuellement à chaque itération sur le format des données.
  const VERSION_FORMAT_DONNEES = 10;

  // Interrupteur global d'affichage des tags — désactivé seulement si
  // l'utilisateur l'a explicitement mis à "0" ; actif par défaut.
  function estActif() {
    return localStorage.getItem(CLE_ACTIF) !== "0";
  }

  // Masquage du tag de club NATIF de GeoGuessr (player-nick_clubTag__) —
  // désactivé par défaut (opt-in), pour ceux qui préfèrent ne voir que
  // le tag de la Ligue sans le doublon du tag de club du jeu.
  function estClubNatifMasque() {
    return localStorage.getItem(CLE_MASQUER_CLUB_NATIF) === "1";
  }

  // Un seul point de vérité pour l'état visuel : une classe sur <body>
  // qui masque tout (badges + carte profil) via CSS. Plus simple et
  // plus robuste que de traquer/masquer chaque élément un par un — ça
  // couvre aussi automatiquement les éléments injectés après coup.
  function appliquerEtatActif() {
    document.body.classList.toggle("laligue-masque", !estActif());
    document.body.classList.toggle("laligue-masque-club-natif", estClubNatifMasque());
  }

  function definirActif(valeur) {
    localStorage.setItem(CLE_ACTIF, valeur ? "1" : "0");
    appliquerEtatActif();
    if (valeur) scannerEtInjecter(); // réinjecte immédiatement plutôt que d'attendre la prochaine mutation DOM
  }

  function definirClubNatifMasque(valeur) {
    localStorage.setItem(CLE_MASQUER_CLUB_NATIF, valeur ? "1" : "0");
    appliquerEtatActif();
  }

  // Filtre par équipe sur l'écran de résultats de challenge — activé par
  // défaut (fonctionnalité déjà en place), mais désactivable pour ceux
  // qui trouvent la barre superflue.
  const CLE_FILTRE_RESULTATS_ACTIF = "laligue_filtre_resultats_actif";

  function estFiltreResultatsActif() {
    return localStorage.getItem(CLE_FILTRE_RESULTATS_ACTIF) !== "0";
  }

  function definirFiltreResultatsActif(valeur) {
    localStorage.setItem(CLE_FILTRE_RESULTATS_ACTIF, valeur ? "1" : "0");
    if (!valeur && filtreResultatsBarre) {
      // Retrait immédiat, pas seulement une prévention pour la prochaine
      // ouverture — sinon la barre resterait affichée jusqu'au prochain
      // changement de page.
      filtreResultatsBarre.remove();
      filtreResultatsBarre = null;
    } else if (valeur) {
      tenterInjectionFiltreResultats(); // réaffiche tout de suite si on est déjà sur /results/
    }
  }

  // Mode d'affichage de la liste d'équipes dans la vue Ligue : groupées
  // par division (défaut) ou triées alphabétiquement à plat.
  const CLE_MODE_TRI = "laligue_mode_tri";
  function obtenirModeTri() {
    return localStorage.getItem(CLE_MODE_TRI) === "alpha" ? "alpha" : "division";
  }
  function definirModeTri(mode) {
    localStorage.setItem(CLE_MODE_TRI, mode);
  }

  // Etat replié/déplié de chaque division dans la vue groupée — mémorisé
  // pour ne pas avoir à tout redéplier à chaque ouverture.
  const CLE_DIVISIONS_REPLIEES = "laligue_divisions_repliees";
  function obtenirDivisionsRepliees() {
    try {
      return new Set(JSON.parse(localStorage.getItem(CLE_DIVISIONS_REPLIEES)) || []);
    } catch (e) {
      return new Set();
    }
  }
  function basculerDivisionRepliee(cle) {
    const repliees = obtenirDivisionsRepliees();
    if (repliees.has(cle)) repliees.delete(cle);
    else repliees.add(cle);
    localStorage.setItem(CLE_DIVISIONS_REPLIEES, JSON.stringify([...repliees]));
    return repliees;
  }

  // ---------------------------------------------------------------
  // Drapeaux des joueurs (pour la composition d'équipe dans la modale)
  // ---------------------------------------------------------------
  // Pas de donnée pays dans l'API de la Ligue — on va la chercher directement sur
  // la page de profil de chaque joueur (elle y affiche déjà un drapeau
  // natif, repéré dès le tout premier extrait HTML fourni). Mis en
  // cache 7 jours (le pays d'un joueur ne change presque jamais) pour
  // ne pas re-télécharger sa page à chaque ouverture de la fiche club.
  const CLE_CACHE_DRAPEAUX = "laligue_drapeaux_cache_v2";
  const DUREE_CACHE_DRAPEAUX_MS = 7 * 24 * 60 * 60 * 1000;

  function lireCacheDrapeaux() {
    try {
      return JSON.parse(localStorage.getItem(CLE_CACHE_DRAPEAUX)) || {};
    } catch (e) {
      return {};
    }
  }

  function ecrireCacheDrapeaux(cache) {
    try {
      localStorage.setItem(CLE_CACHE_DRAPEAUX, JSON.stringify(cache));
    } catch (e) {
      // pas grave si indisponible
    }
  }

  async function obtenirDrapeauJoueur(userId) {
    if (!userId) return null;
    const cache = lireCacheDrapeaux();
    const entree = cache[userId];
    if (entree && Date.now() - entree.horodatage < DUREE_CACHE_DRAPEAUX_MS) {
      return entree.pays ? entree : null;
    }
    try {
      const reponse = await mettreEnFile(() => fetch(`https://www.geoguessr.com/user/${userId}`), true);
      const texte = await reponse.text();
      // Le HTML brut ne contient PAS encore la balise <img> du drapeau
      // (rendue par React après coup), mais le JSON de données de la
      // page (__NEXT_DATA__) contient déjà countryCode en clair.
      const match = texte.match(/"countryCode":"([a-zA-Z-]{2,6})"/);
      const pays = match ? match[1].toLowerCase() : null;
      // On met en cache MÊME un échec (pays: null), pour ne pas
      // retenter cette requête à chaque ouverture de fiche.
      const resultat = { pays, horodatage: Date.now() };
      // Relecture juste avant écriture : une autre requête en parallèle
      // a pu ajouter son entrée depuis notre lecture initiale, et
      // réécrire notre copie périmée la ferait disparaître (entrée perdue
      // = requête refaite inutilement plus tard).
      const cacheAJour = lireCacheDrapeaux();
      cacheAJour[userId] = resultat;
      ecrireCacheDrapeaux(cacheAJour);
      return resultat.pays ? resultat : null;
    } catch (e) {
      return null; // page privée, joueur supprimé, hors ligne... on se passe juste du drapeau
    }
  }

  let LANGUE =
    localStorage.getItem(CLE_LANGUE) ||
    (navigator.language && navigator.language.toLowerCase().startsWith("fr") ? "fr" : "en");

  function t(cle, ...args) {
    const valeur = (TRADUCTIONS[LANGUE] || TRADUCTIONS.fr)[cle];
    return typeof valeur === "function" ? valeur(...args) : valeur;
  }

  function definirLangue(langue) {
    LANGUE = langue;
    localStorage.setItem(CLE_LANGUE, langue);
  }

  function lireCache() {
    try {
      const brut = localStorage.getItem(CLE_CACHE_DONNEES);
      if (!brut) return null;
      const parsed = JSON.parse(brut);
      if (parsed.versionFormat !== VERSION_FORMAT_DONNEES) return null;
      if (!parsed.horodatage || Date.now() - parsed.horodatage > DUREE_CACHE_MS) return null;
      return parsed.donnees;
    } catch (e) {
      return null;
    }
  }

  function ecrireCache(donnees) {
    try {
      localStorage.setItem(
        CLE_CACHE_DONNEES,
        JSON.stringify({ horodatage: Date.now(), versionFormat: VERSION_FORMAT_DONNEES, donnees })
      );
    } catch (e) {
      // pas grave si indisponible, on refetchera simplement la prochaine fois
    }
  }

  // ---------------------------------------------------------------
  // Couche de données
  // ---------------------------------------------------------------
  // ⚠️ SEULE CETTE FONCTION (et les constantes juste en dessous) devra
  // être réajustée si l'organisation change d'API ou de structure de
  // données. Tout le reste du script ne connaît que le format
  // normalisé retourné ici.
  //
  // Format attendu en sortie :
  // {
  //   equipes: {
  //     [equipeId]: {
  //       nom: "Les Fous du Bus",
  //       tag: "LFDB",
  //       division: 3,             // champ "division" de l'API (1 à 5), null si absent
  //       couleurFond: "#F7931E",  // "backColor"
  //       couleurTexte: "#FFFFFF", // "frontColor"
  //       logoUrl: "https://www.laliguegeoguessr.fr/resources/logos/S12/LFDB.png",
  //       membres: [{ pseudo, userId, twitch }],
  //     },
  //     ...
  //   },
  //   joueurs: {
  //     // clé = ID GeoGuessr ("geoguessrId" de l'API)
  //     [geoguessrId]: { equipeId: "...", classement: { place: 3, points: "57" } | null },
  //   },
  //   // Repli par pseudo — DERNIER RECOURS uniquement (voir
  //   // resoudreEquipeJoueur). Le pseudo n'est pas fiable comme
  //   // identifiant à l'échelle de GeoGuessr, mais reste nécessaire pour
  //   // certains contextes (liste d'amis) où l'ID n'est pas exposé côté
  //   // client. Exclut les pseudos ambigus entre deux équipes de la
  //   // Ligue elle-même — voir equipeVuePourPseudo.
  //   joueursParPseudo: {
  //     [pseudoExact]: { equipeId: "...", classement: { place, points } | null },
  //   },
  // }

  // Source officielle depuis la S12 : l'API de l'organisation, qui
  // remplace l'ancien Google Sheet (parsing par index de colonnes,
  // fragile au moindre déplacement). Format d'une équipe :
  //   { id, season, name, tag, division, backColor, frontColor,
  //     currentRanking, players: [ { geoguessrId, name, socialLinks } ] }
  // Une division précise peut être demandée avec &division=N, mais on
  // charge tout d'un coup : le volume est modeste et le cache évite de
  // refaire l'appel.
  const SAISON = 12;
  const URL_API_EQUIPES = `https://www.laliguegeoguessr.fr/api/teams?season=${SAISON}`;

  // Les logos ne sont pas dans l'API : ils suivent une convention de
  // nommage par tag, confirmée par l'organisation (28/08/2026).
  function urlLogoEquipe(tag) {
    return `https://www.laliguegeoguessr.fr/resources/logos/S${SAISON}/${encodeURIComponent(tag)}.png`;
  }

  // socialLinks est un tableau pouvant contenir des chaînes vides, ou des
  // liens qui ne sont PAS Twitch (YouTube par exemple) — on ne retient
  // que le premier lien Twitch, et seulement s'il passe urlSure().
  function extraireLienTwitch(socialLinks) {
    if (!Array.isArray(socialLinks)) return null;
    const lien = socialLinks.find((l) => typeof l === "string" && /twitch\.tv\//i.test(l));
    return lien ? urlSure(lien) : null;
  }

  // Compatibilité Tampermonkey / Greasemonkey ancien ET nouveau : Tampermonkey
  // (et l'ancien Greasemonkey) exposent `GM_xmlhttpRequest`, mais Greasemonkey
  // a renommé cette fonction en `GM.xmlHttpRequest` (espace de nommage "GM."
  // + majuscule à "Http") et ne fournit plus l'ancien nom du tout. On essaie
  // l'ancien nom en premier (le plus répandu), puis le nouveau en repli.
  function requeteXhr(details) {
    if (typeof GM_xmlhttpRequest === "function") {
      return GM_xmlhttpRequest(details);
    }
    if (typeof GM !== "undefined" && typeof GM.xmlHttpRequest === "function") {
      return GM.xmlHttpRequest(details);
    }
    // Aucune des deux API n'est disponible — on prévient clairement plutôt
    // que de planter silencieusement plus loin dans la Promise.
    console.error("[LLGG] Ni GM_xmlhttpRequest ni GM.xmlHttpRequest ne sont disponibles.");
    if (details.onerror) details.onerror(new Error("GM_xmlhttpRequest indisponible"));
  }

  // L'endpoint de classement, appelé sans paramètre "day", renvoie TOUTES
  // les journées à la suite (une entrée par équipe ET par journée) — on
  // ne garde, pour chaque équipe, que la journée la plus récente, qui
  // porte le classement cumulé en cours. Un seul appel couvre les cinq
  // divisions. Champs : day (nombre), place et points (TEXTE, virgule
  // décimale à la française).
  const URL_API_CLASSEMENTS = `https://www.laliguegeoguessr.fr/api/rankings/season?season=${SAISON}`;

  // avecPoints = false dans l'infobulle des badges, volontairement
  // compacte (place seule) ; la carte de profil garde les points.
  function formaterClassement(classement, avecPoints = true) {
    if (!classement || !Number.isInteger(classement.place)) return null;
    return t("classementPoints", classement.place, avecPoints ? classement.points : "");
  }

  function construireClassementsParTag(entreesApi) {
    const parTag = {};
    (Array.isArray(entreesApi) ? entreesApi : []).forEach((entree) => {
      const tag = String(entree.teamTag || "").trim();
      const jour = Number(entree.day);
      const place = parseInt(entree.place, 10);
      if (!tag || !Number.isFinite(jour) || !Number.isInteger(place) || place < 1) return;
      if (parTag[tag] && parTag[tag].jour >= jour) return; // on a déjà une journée plus récente
      parTag[tag] = { jour, place, points: String(entree.points ?? "").trim() };
    });
    return parTag;
  }

  function construireDonneesDepuisApi(equipesApi, classementsParTag = {}) {
    const equipes = {};
    const joueurs = {};
    // Repli par pseudo — dernier recours UNIQUEMENT (voir
    // resoudreEquipeJoueur) pour les contextes où GeoGuessr n'expose
    // l'ID nulle part côté client (liste d'amis notamment, vérifié
    // exhaustivement le 16/08/2026 : ni requête réseau, ni WebSocket, ni
    // props/state React, ni attribut DOM ne le contient à cet endroit).
    // Risque résiduel accepté : un joueur hors Ligue portant EXACTEMENT
    // le même pseudo qu'un joueur de la Ligue serait tagué à tort. On
    // limite ce risque en désactivant le repli dès qu'un pseudo est
    // partagé par deux équipes DIFFÉRENTES de la Ligue elle-même.
    const joueursParPseudo = {};
    const equipeVuePourPseudo = {};

    (Array.isArray(equipesApi) ? equipesApi : []).forEach((equipeApi) => {
      const tag = String(equipeApi.tag || "").trim();
      const nom = String(equipeApi.name || "").trim();
      if (!tag || !nom) return; // entrée inexploitable, on l'ignore plutôt que d'afficher un badge vide

      const equipeId = tag.toLowerCase();
      const division = Number.isInteger(equipeApi.division) ? equipeApi.division : null;
      // Classement : source prioritaire = l'API des classements (voir
      // construireClassementsParTag). currentRanking de l'API des équipes
      // n'est qu'un repli — constaté le 19/10/2026 : il reste à 0 pour
      // TOUTES les équipes de D2 et D4, alors que leur classement existe
      // bien côté organisation.
      // Stocké BRUT ({ place, points }) et formaté seulement à l'affichage
      // (voir formaterClassement) : ces données partent dans le cache, et
      // un texte déjà traduit y resterait figé dans l'ancienne langue
      // après un changement FR/EN.
      const infoClassement = classementsParTag[tag];
      const classement = infoClassement
        ? { place: infoClassement.place, points: infoClassement.points }
        : equipeApi.currentRanking > 0
          ? { place: equipeApi.currentRanking, points: "" }
          : null;

      equipes[equipeId] = {
        nom,
        tag,
        division,
        couleurFond: equipeApi.backColor || "#888888",
        couleurTexte: equipeApi.frontColor || "#111111",
        logoUrl: urlLogoEquipe(tag),
        // Aussi porté par l'équipe (et pas seulement par chaque joueur) :
        // la vue Ligue ouvre la fiche directement depuis l'équipe.
        classement,
        membres: [], // rempli juste en dessous, pour la modale de détail d'équipe
      };

      (Array.isArray(equipeApi.players) ? equipeApi.players : []).forEach((joueurApi) => {
        const pseudo = String(joueurApi.name || "").trim();
        const userId = String(joueurApi.geoguessrId || "").trim() || null;
        if (!pseudo && !userId) return;

        const entree = { equipeId, classement, userId };
        if (userId) joueurs[userId] = entree;

        if (pseudo) {
          if (equipeVuePourPseudo[pseudo] === undefined) {
            equipeVuePourPseudo[pseudo] = equipeId;
            joueursParPseudo[pseudo] = entree;
          } else if (equipeVuePourPseudo[pseudo] !== equipeId) {
            // Homonyme entre deux équipes de la Ligue : on retire toute
            // résolution par ce pseudo, plutôt que de risquer d'afficher
            // le tag de la mauvaise équipe.
            delete joueursParPseudo[pseudo];
            equipeVuePourPseudo[pseudo] = "AMBIGU";
          }
        }

        equipes[equipeId].membres.push({
          pseudo,
          userId,
          twitch: extraireLienTwitch(joueurApi.socialLinks),
        });
      });
    });

    return { equipes, joueurs, joueursParPseudo };
  }

  function requeteJson(url) {
    return new Promise((resolve, reject) => {
      requeteXhr({
        method: "GET",
        url,
        onload: (res) => {
          try {
            resolve(JSON.parse(res.responseText));
          } catch (e) {
            reject(e);
          }
        },
        onerror: reject,
      });
    });
  }

  async function fetchDonneesLigueDepuisSource() {
    // Lancés en parallèle. Les équipes sont indispensables (sans elles,
    // aucun badge) ; le classement est un bonus : s'il échoue, on
    // continue sans, plutôt que de tout faire tomber.
    const [equipesApi, classementsApi] = await Promise.all([
      requeteJson(URL_API_EQUIPES),
      requeteJson(URL_API_CLASSEMENTS).catch((e) => {
        console.warn("[LLGG] Classements indisponibles (pas grave, juste pas affichés) :", e);
        return [];
      }),
    ]);
    try {
      return construireDonneesDepuisApi(equipesApi, construireClassementsParTag(classementsApi));
    } catch (e) {
      console.error("[LLGG] Échec du parsing de l'API de la Ligue:", e);
      throw e;
    }
  }

  let donneesLigueEnMemoire = null;
  let promesseChargement = null;

  async function obtenirDonneesLigue() {
    if (donneesLigueEnMemoire) return donneesLigueEnMemoire;

    const enCache = lireCache();
    if (enCache) {
      donneesLigueEnMemoire = enCache;
      return enCache;
    }

    if (!promesseChargement) {
      promesseChargement = fetchDonneesLigueDepuisSource()
        .then((donnees) => {
          donneesLigueEnMemoire = donnees;
          ecrireCache(donnees);
          logDebug(
            `Données chargées : ${Object.keys(donnees.equipes).length} équipes, ${Object.keys(donnees.joueurs).length} joueurs par ID, ${Object.keys(donnees.joueursParPseudo).length} joueurs par pseudo (repli).`
          );
          return donnees;
        })
        .catch((e) => {
          console.warn("[LLGG] Échec du chargement des données:", e);
          promesseChargement = null; // permet de réessayer au prochain appel
          return null;
        });
    }
    return promesseChargement;
  }

  // ---------------------------------------------------------------
  // Résolution joueur -> équipe
  // ---------------------------------------------------------------
  // ID GeoGuessr prioritaire — toujours fiable, jamais d'ambiguïté.
  // Repli par pseudo en DERNIER RECOURS uniquement, quand l'ID n'est
  // vraiment pas extractible du DOM (confirmé pour la liste d'amis le
  // 16/08/2026, après vérification qu'aucune requête réseau, WebSocket,
  // prop/state React ni attribut DOM ne l'expose à cet endroit).
  // ⚠️ Risque résiduel ASSUMÉ : un joueur hors Ligue portant EXACTEMENT
  // le même pseudo qu'un joueur de la Ligue serait tagué à tort. Réduit
  // (mais pas éliminé) par la garde anti-homonyme entre équipes de la
  // Ligue dans construireDonneesDepuisApi — voir joueursParPseudo.
  // ---------------------------------------------------------------
  // Mémoire des pseudos actuels (apprise automatiquement)
  // ---------------------------------------------------------------
  // Chaque fois qu'un joueur de la Ligue est résolu via son ID (fiable —
  // résultats, classement Duels, page de profil, party...), on retient
  // son pseudo ACTUEL à ce moment-là. Cette mémoire sert ensuite de
  // repli prioritaire dans les contextes où l'ID n'est pas extractible
  // (liste d'amis) — plus fiable que le pseudo figé de l'API, puisqu'il
  // reflète ce qui est réellement affiché en jeu. Alimentée gratuitement
  // par la simple navigation normale sur le site, sans action dédiée.
  const CLE_PSEUDOS_APPRIS = "laligue_pseudos_appris";

  // Lecture JSON mémoïsée : ces caches sont consultés à chaque
  // résolution par pseudo, potentiellement des centaines de fois par
  // scan sur un gros classement. On ne re-parse que si la chaîne brute a
  // changé (écriture par nous, ou par un autre onglet) — le getItem seul
  // reste nécessaire pour rester à jour, mais c'est le JSON.parse qui
  // coûtait. ⚠️ La valeur renvoyée est partagée : toujours la copier
  // avant de la modifier (voir les fonctions d'écriture).
  const memoJson = {};
  function lireJsonStockage(cle) {
    let brut = null;
    try {
      brut = localStorage.getItem(cle);
    } catch (e) {
      return {};
    }
    const memo = memoJson[cle];
    if (memo && memo.brut === brut) return memo.valeur;
    let valeur = {};
    try {
      valeur = JSON.parse(brut) || {};
    } catch (e) {
      valeur = {};
    }
    memoJson[cle] = { brut, valeur };
    return valeur;
  }

  function lirePseudosAppris() {
    return lireJsonStockage(CLE_PSEUDOS_APPRIS);
  }

  // Regroupée en une seule par frame (même principe que grouperParFrame
  // pour l'observer) : au premier chargement, de nombreux joueurs
  // peuvent être appris quasi simultanément — sans ce regroupement,
  // chacun déclencherait sa propre remise à zéro, se chevauchant avec le
  // traitement d'autres joueurs encore en cours.
  const reinitialiserWrappersTraite = grouperParFrame(() => {
    document
      .querySelectorAll(
        '[class*="player-nick_wrapper"][data-laligue-traite], [class*="activity-card_groupActor__"][data-laligue-traite]'
      )
      .forEach((w) => {
      w.querySelector(":scope > .laligue-badge")?.remove();
      delete w.dataset.laligueTraite;
      delete w.dataset.laligueIdentite;
    });
  });

  function enregistrerPseudoAppris(pseudo, userId) {
    if (!pseudo || !userId) return;
    const cache = { ...lirePseudosAppris() }; // copie : la lecture est mémoïsée et partagée
    if (cache[pseudo] === userId) return; // déjà su, rien à faire

    cache[pseudo] = userId;
    try {
      localStorage.setItem(CLE_PSEUDOS_APPRIS, JSON.stringify(cache));
    } catch (e) {
      // pas grave si indisponible
    }

    // Nouvelle info : un wrapper avec ce pseudo (ex: dans la liste
    // d'amis) a pu être scanné et marqué "pas d'équipe" avant qu'on le
    // sache — on force un nouveau passage pour rattraper ce cas. Le
    // vrai garde-fou anti-doublon est dans traiterWrapper (retrait
    // inconditionnel avant insertion) ; ce regroupement par frame limite
    // juste le nombre de passages inutiles.
    reinitialiserWrappersTraite();
  }

  function obtenirPseudoAppris(pseudo) {
    return lirePseudosAppris()[pseudo] || null;
  }

  // ---------------------------------------------------------------
  // Vérification active des correspondances par pseudo
  // ---------------------------------------------------------------
  // Un tag posé par pseudo est une DÉDUCTION (le joueur affiché ici
  // pourrait être un homonyme hors Ligue). Plutôt que de le présenter
  // avec une incertitude visuelle, on le vérifie activement en tâche de
  // fond : on va voir la vraie page de profil du joueur de la Ligue
  // qu'on pense reconnaître, et on compare son pseudo ACTUEL à celui
  // affiché. Si ça ne correspond pas, ce pseudo est mémorisé comme "ne
  // doit jamais avoir ce tag" et le badge est retiré partout.
  const CLE_PSEUDOS_INVALIDES = "laligue_pseudos_invalides";
  const CLE_PSEUDOS_VERIFIES = "laligue_pseudos_verifies_le"; // pseudo -> horodatage de dernière vérification
  const DUREE_VERIF_MS = 24 * 60 * 60 * 1000; // 24h avant de revérifier un pseudo déjà validé

  function lirePseudosInvalides() {
    return lireJsonStockage(CLE_PSEUDOS_INVALIDES);
  }

  function pseudoEstInvalide(pseudo) {
    return !!lirePseudosInvalides()[pseudo];
  }

  function marquerPseudoInvalide(pseudo) {
    const cache = { ...lirePseudosInvalides() }; // copie : la lecture est mémoïsée et partagée
    if (cache[pseudo]) return;
    cache[pseudo] = true;
    try {
      localStorage.setItem(CLE_PSEUDOS_INVALIDES, JSON.stringify(cache));
    } catch (e) {
      // pas grave si indisponible
    }
    // Retire le badge (et son icône) partout où ce pseudo était affiché par déduction.
    document
      .querySelectorAll(
        '[class*="player-nick_wrapper"][data-laligue-identite], [class*="activity-card_groupActor__"][data-laligue-identite]'
      )
      .forEach((w) => {
      if (w.dataset.laligueIdentite === pseudo) {
        w.querySelector(":scope > .laligue-badge")?.remove();
        w.querySelector(":scope > .laligue-badge-signaler")?.remove();
        delete w.dataset.laligueTraite;
        delete w.dataset.laligueIdentite;
      }
    });
  }

  function pseudoDejaVerifieRecemment(pseudo) {
    try {
      const cache = JSON.parse(localStorage.getItem(CLE_PSEUDOS_VERIFIES)) || {};
      return cache[pseudo] && Date.now() - cache[pseudo] < DUREE_VERIF_MS;
    } catch (e) {
      return false;
    }
  }

  function marquerPseudoVerifie(pseudo) {
    try {
      const cache = JSON.parse(localStorage.getItem(CLE_PSEUDOS_VERIFIES)) || {};
      cache[pseudo] = Date.now();
      localStorage.setItem(CLE_PSEUDOS_VERIFIES, JSON.stringify(cache));
    } catch (e) {
      // pas grave si indisponible
    }
  }

  // Vide entièrement la mémoire pseudo (apprentissage + liste noire +
  // horodatages de vérification) — utile en cas de fausse manipulation
  // (signalement accidentel d'un vrai joueur, par exemple).
  function reinitialiserMemoirePseudos() {
    localStorage.removeItem(CLE_PSEUDOS_APPRIS);
    localStorage.removeItem(CLE_PSEUDOS_INVALIDES);
    localStorage.removeItem(CLE_PSEUDOS_VERIFIES);
    // Force un nouveau passage sur tout, badges + icônes de signalement
    // retirés au passage pour repartir propre.
    document
      .querySelectorAll(
        '[class*="player-nick_wrapper"][data-laligue-traite], [class*="activity-card_groupActor__"][data-laligue-traite]'
      )
      .forEach((w) => {
      w.querySelector(":scope > .laligue-badge")?.remove();
      w.querySelector(":scope > .laligue-badge-signaler")?.remove();
      delete w.dataset.laligueTraite;
      delete w.dataset.laligueIdentite;
    });
  }

  // Va chercher le pseudo ACTUEL du joueur candidat sur sa vraie page de
  // profil (même technique que pour le pays/l'ID perso : le JSON de la
  // page contient déjà "nick":"..." en clair). Si ça ne correspond pas
  // au pseudo affiché, on écarte définitivement ce faux positif.
  async function verifierCorrespondancePseudo(pseudoAffiche, idCandidat) {
    if (!idCandidat) return; // pas d'ID candidat à vérifier (l'API ne l'avait pas non plus)
    if (pseudoDejaVerifieRecemment(pseudoAffiche)) return;
    marquerPseudoVerifie(pseudoAffiche); // avant l'appel réseau, pour ne pas relancer en boucle en attendant la réponse

    try {
      const reponse = await mettreEnFile(() => fetch(`https://www.geoguessr.com/user/${idCandidat}`));
      const texte = await reponse.text();
      const match = texte.match(/"nick":"([^"]+)"/);
      const nickReel = match ? match[1] : null;
      if (nickReel && nickReel !== pseudoAffiche) {
        logDebug(
          `Vérification pseudo : "${pseudoAffiche}" affiché, mais le joueur ${idCandidat} s'appelle en réalité "${nickReel}" — faux positif écarté`
        );
        marquerPseudoInvalide(pseudoAffiche);
      } else {
        logDebug(`Vérification pseudo : "${pseudoAffiche}" confirmé pour ${idCandidat}`);
      }
    } catch (e) {
      // pas grave, on retentera dans 24h
    }
  }

  function resoudreEquipeJoueur(donnees, { userId, pseudo }) {
    if (!donnees) return null;

    let entree = null;
    let fiabilite = "id"; // "id" = certain (lien /user/ vérifié), "pseudo" = probable, à vérifier
    if (userId && donnees.joueurs[userId]) {
      entree = donnees.joueurs[userId];
    } else if (pseudo && !pseudoEstInvalide(pseudo)) {
      // Repli par pseudo. Si un ID est connu (lien /user/ présent) mais
      // n'est pas celui du joueur trouvé par pseudo, c'est un HOMONYME :
      // on n'en tient pas compte (bug du 22/10/2026, deux "ferreiro" sur
      // le classement Duels). Seul cas où le pseudo reste utile malgré un
      // ID : un inscrit dont l'API ne fournit pas d'ID.
      const compatible = (candidat) => !userId || !candidat.userId || candidat.userId === userId;
      // Priorité au pseudo APPRIS (vu récemment ailleurs avec un ID
      // fiable) avant le repli sur le pseudo figé de l'API, qui peut
      // être périmé si le joueur a changé de pseudo depuis son
      // inscription. Les deux restent une DÉDUCTION, vérifiée a
      // posteriori en tâche de fond — voir verifierCorrespondancePseudo.
      const idAppris = obtenirPseudoAppris(pseudo);
      if (idAppris && donnees.joueurs[idAppris] && compatible(donnees.joueurs[idAppris])) {
        entree = donnees.joueurs[idAppris];
        fiabilite = "pseudo";
      } else if (donnees.joueursParPseudo[pseudo] && compatible(donnees.joueursParPseudo[pseudo])) {
        entree = donnees.joueursParPseudo[pseudo];
        fiabilite = "pseudo";
      }
    }
    if (!entree) return null;

    const equipe = donnees.equipes[entree.equipeId];
    if (!equipe) return null;

    return { ...equipe, classement: entree.classement || null, fiabilite, idCandidat: entree.userId || null };
  }

  // Luminance perçue approximative — détermine si une couleur de fond
  // est "claire" (nécessite un texte sombre pour rester lisible), peu
  // importe ce que dit le champ frontColor de l'API. Filet de
  // sécurité invisible pour 95% des équipes, qui évite un texte blanc
  // sur fond clair illisible si une équipe rentre une couleur pastel.
  function couleurEstClaire(hex) {
    const m = (hex || "").replace("#", "").match(/^([0-9a-f]{2})([0-9a-f]{2})([0-9a-f]{2})$/i);
    if (!m) return false;
    const [, r, g, b] = m.map((v, i) => (i === 0 ? v : parseInt(v, 16)));
    const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
    return luminance > 0.6;
  }

  function couleurTexteSure(couleurFond, couleurTexteSouhaite) {
    return couleurEstClaire(couleurFond) ? "#111111" : couleurTexteSouhaite || "#ffffff";

  // "javascript:..." placée dans un href s'exécuterait au clic dans le
  // contexte de la page GeoGuessr (accès aux cookies, à la session...).
  // On n'accepte donc QUE http(s), et on ignore silencieusement le reste
  // plutôt que de faire confiance à la source.
  function urlSure(url) {
    if (typeof url !== "string") return null;
    const propre = url.trim();
    return /^https?:\/\//i.test(propre) ? propre : null;
  }

  function creerCarteProfil(equipe) {
    ajouterStylesCarteProfil();

  }

  // Ancre confirmée par inspection DOM (17/08/2026, version stabilisée
  // après plusieurs itérations) : priorité au badge de club natif lui-
  // même (`profile-v2_clubLink__`) — carte positionnée juste au-dessus,
  // alignée à gauche avec lui. Repli sur le bandeau d'en-tête ENTIER
  // (`profile-v2_header__`, toujours présent qu'il y ait un club ou non)
  // pour les joueurs SANS club natif, où on calcule nous-mêmes le coin
  // haut-droite plutôt que de dépendre d'un sous-conteneur dont la
  // position s'est révélée incohérente selon les profils. Voir
  // positionnerCarteProfilFlottante pour la logique de position, qui
  // diffère selon lequel des deux a été trouvé.
  function trouverAncreClubProfil() {
    return (
      document.querySelector('[class*="profile-v2_clubLink__"]') ||
      document.querySelector('[class*="profile-v2_header__"]')
    );
  }

  let carteProfilEnCours = false; // évite les tentatives concurrentes pendant l'await

  // /me/profile est la page de profil DU JOUEUR CONNECTÉ — pas d'ID dans
  // l'URL contrairement à /user/{id}. Confirmé par test réel (16/08/2026)
  // que le JSON de données embarqué par Next.js contient
  // "userProfile":{"user":{"userId":"..."}}.
  const CLE_MON_ID = "laligue_mon_id";

  function extraireProprIdDepuisPage() {
    const scriptDonnees = document.getElementById("__NEXT_DATA__");
    const texte = scriptDonnees ? scriptDonnees.textContent : document.documentElement.outerHTML;
    const match = texte.match(/"userId":"([0-9a-f]{24})"/);
    // Mémorisé au passage : c'est lui qui identifie l'équipe du joueur
    // connecté pour l'onglet LLGG (voir trouverMonEquipe).
    if (match) localStorage.setItem(CLE_MON_ID, match[1]);
    return match ? match[1] : null;
  }

  // Positionne la carte (hors flux, appartenant à document.body) juste
  // au-dessus de l'élément passé en ancre, en coordonnées "page" (pas
  // "viewport") — ainsi elle défile naturellement avec la page sans
  // avoir besoin d'écouter le scroll, seulement les changements de mise
  // en page (resize, contenu qui bouge).
  // Coin haut-droite du bandeau d'en-tête, avec une marge — reproduit
  // grossièrement l'emplacement du badge de club natif (toujours en haut
  // à droite du bandeau), sans dépendre d'un sous-élément précis.
  function positionnerCarteProfilFlottante(carte, ancreElement) {
    const rect = ancreElement.getBoundingClientRect();
    const estBadgeClub = ancreElement.className && /profile-v2_clubLink__/.test(ancreElement.className);
    if (estBadgeClub) {
      // Comportement d'origine, qui fonctionnait bien : juste au-dessus
      // du badge de club natif, aligné à gauche avec lui.
      carte.style.left = `${rect.left + window.scrollX}px`;
      carte.style.top = `${rect.top + window.scrollY - carte.offsetHeight - 8}px`;
    } else {
      // Repli sans club natif : coin haut-droite du bandeau entier,
      // décalage vertical réglé empiriquement (17/08/2026) pour rester
      // cohérent avec le reste de l'en-tête à cet endroit.
      carte.style.left = `${rect.right + window.scrollX - carte.offsetWidth - 16}px`;
      carte.style.top = `${rect.top + window.scrollY + 55}px`;
    }
  }

  let carteProfilFlottante = null; // référence à la carte actuellement affichée, hors DOM GeoGuessr

  async function tenterInjectionCarteProfilEntete() {
    if (!estActif()) return;

    // Cache la carte (sans la retirer du DOM, juste invisible) tant
    // qu'une fenêtre modale native de GeoGuessr est ouverte par-dessus
    // (ex: "Show stats") — `overlay-modal_contentContainer__` semble être
    // leur système de modale générique (même préfixe que leur propre
    // bouton de fermeture), donc ça couvre potentiellement toutes leurs
    // fenêtres natives, pas seulement celle des stats. Réapparaît toute
    // seule dès que la fenêtre native se ferme, sans avoir à la
    // reconstruire.
    if (carteProfilFlottante) {
      const modaleNativeOuverte = !!document.querySelector('[class*="overlay-modal_contentContainer__"]');
      carteProfilFlottante.style.visibility = modaleNativeOuverte ? "hidden" : "";
    }

    if (carteProfilEnCours) return;

    const matchUrl = location.pathname.match(/^\/user\/([a-zA-Z0-9]+)/);
    let userId = null;
    if (matchUrl) {
      userId = matchUrl[1];
    } else if (location.pathname.startsWith("/me/profile")) {
      userId = extraireProprIdDepuisPage();
    } else {
      // Plus sur une page de profil : on retire la carte flottante
      // restante s'il y en a une (sinon elle resterait affichée en
      // naviguant ailleurs, puisqu'elle vit dans document.body, hors de
      // portée du routage de GeoGuessr).
      if (carteProfilFlottante) {
        carteProfilFlottante.remove();
        carteProfilFlottante = null;
      }
      return;
    }
    if (!userId) return;

    const ancre = trouverAncreClubProfil();
    if (!ancre) return; // pas encore rendu par React, on retentera au prochain scan

    // Déjà affichée pour CE joueur : juste re-synchroniser la position
    // (leur badge de club a pu bouger — resize, contenu qui charge...).
    if (carteProfilFlottante && carteProfilFlottante.dataset.userId === userId) {
      positionnerCarteProfilFlottante(carteProfilFlottante, ancre);
      return;
    }

    // Nouveau joueur (changement de profil en SPA) : on retire l'ancienne
    // carte avant de reconstruire, plutôt que d'empiler.
    if (carteProfilFlottante) {
      carteProfilFlottante.remove();
      carteProfilFlottante = null;
    }

    carteProfilEnCours = true;
    try {
      const donnees = await obtenirDonneesLigue();
      const equipe = resoudreEquipeJoueur(donnees, { userId });
      if (!equipe) return;

      const pseudoProfil = document.querySelector('[data-qa="user-card-title"] h1')?.textContent.trim();
      if (pseudoProfil) enregistrerPseudoAppris(pseudoProfil, userId);

      const carte = creerCarteProfil(equipe);
      carte.classList.add("laligue-carte-profil-flottante");
      carte.dataset.userId = userId;
      document.body.appendChild(carte);
      positionnerCarteProfilFlottante(carte, ancre);
      carteProfilFlottante = carte;
    } finally {
      carteProfilEnCours = false;
    }
  }

  // Le redimensionnement de fenêtre ne déclenche aucune mutation DOM,
  // donc le scan habituel (MutationObserver) ne le détecterait jamais —
  // écoute dédiée pour resynchroniser la position dans ce cas précis.
  if (!SUR_SITE_LIGUE) {
    window.addEventListener(
      "resize",
      grouperParFrame(() => {
        if (carteProfilFlottante) {
          const ancre = trouverAncreClubProfil();
          if (ancre) positionnerCarteProfilFlottante(carteProfilFlottante, ancre);
        }
      })
    );
  }

      tagEl.className = "laligue-modal-tag";

      tagEl.style.background = equipe.couleurFond;
      tagEl.style.color = couleurTexteSure(equipe.couleurFond, equipe.couleurTexte);
      nomSpan.appendChild(tagEl);
    }

    const divisionEl = creerTexteDivision(equipe.division, "laligue-modal-division");
    // Raccourcis vers les pages de la division sur le site de
    // l'organisation — mêmes liens que les en-têtes de division de la vue
    // Ligue (sélection de la division à l'arrivée, voir demarrerSiteLigue).
    if (divisionEl) {
      // Place au classement juste après "Division X" (sans les points,
      // comme l'infobulle), avant les icônes.
      const place = formaterClassement(equipe.classement, false);
      if (place) {
        const placeEl = document.createElement("span");
        placeEl.className = "laligue-modal-place";
        placeEl.textContent = `· ${place}`;
        divisionEl.appendChild(placeEl);
      }
      [
        { url: urlLigueStats(equipe.division), titre: t("lienStatsTitre", equipe.division), svg: SVG_STATS },
        { url: urlLigueClassement(equipe.division), titre: t("lienClassementTitre", equipe.division), svg: SVG_CLASSEMENT },
      ].forEach(({ url, titre, svg }) => {
        const lien = document.createElement("a");
        lien.className = "laligue-modal-division-lien";
        lien.href = url;
        lien.target = "_blank";
        lien.rel = "noopener noreferrer";
        lien.title = titre;
        lien.setAttribute("aria-label", titre);
        lien.innerHTML = svg;
        lien.addEventListener("click", (e) => e.stopPropagation());
        divisionEl.appendChild(lien);
      });
    }

    const souscription = document.createElement("div");
    souscription.className = "laligue-modal-souscription";
    souscription.textContent = t("nbJoueurs", (equipe.membres || []).length);

    texteHeader.appendChild(nomSpan);
    if (divisionEl) texteHeader.appendChild(divisionEl);
    texteHeader.appendChild(souscription);

    header.appendChild(logo);
    header.appendChild(texteHeader);

    const liste = document.createElement("ul");
    liste.className = "laligue-modal-membres";
    (equipe.membres || []).forEach((membre) => {
      const li = document.createElement("li");
      const lien = document.createElement("a");
      lien.href = membre.userId ? `https://www.geoguessr.com/user/${membre.userId}` : "#";
      lien.target = "_blank";
      lien.rel = "noopener";

      const drapeau = document.createElement("img");
      drapeau.className = "laligue-modal-drapeau";
      drapeau.alt = "";

      const texte = document.createElement("span");
      texte.textContent = membre.pseudo;

      lien.appendChild(drapeau);
      lien.appendChild(texte);
      li.appendChild(lien);
      liste.appendChild(li);

      if (membre.userId) {
        obtenirDrapeauJoueur(membre.userId).then((info) => {
          if (!info || !info.pays) {
            drapeau.remove(); // aucun drapeau trouvé, certain — on ne laisse pas un squelette tourner pour rien
            return;
          }
          // flagcdn.com gère aussi les nations infranationales de
          // GeoGuessr (gb-eng, gb-sct, gb-wls...), pas juste les codes
          // ISO à 2 lettres classiques.
          drapeau.src = `https://flagcdn.com/w40/${info.pays}.png`;
          drapeau.title = info.pays.toUpperCase();
          drapeau.onerror = () => drapeau.remove(); // code non reconnu par flagcdn — pas d'icône cassée
          drapeau.classList.add("laligue-drapeau-charge");
        });
      } else {
        drapeau.remove(); // pas d'ID pour ce membre, on ne pourra jamais avoir son drapeau
      }

      // Icône Twitch — élément SÉPARÉ du lien principal (deux <a> dans le
      // même <li>). Depuis la S12, le lien vient directement de l'API
      // (socialLinks), donc plus aucune requête ni attente : on sait tout
      // de suite si ce joueur en a un. Indépendant de l'ID GeoGuessr,
      // contrairement au drapeau.
      if (membre.twitch) {
        const lienTwitch = document.createElement("a");
        lienTwitch.className = "laligue-modal-twitch";
        lienTwitch.href = membre.twitch;
        lienTwitch.target = "_blank";
        lienTwitch.rel = "noopener noreferrer";
        lienTwitch.title = `Twitch — ${membre.pseudo}`;
        lienTwitch.addEventListener("click", (e) => e.stopPropagation());
        lienTwitch.innerHTML = `<svg viewBox="0 0 24 24" fill="currentColor" xmlns="http://www.w3.org/2000/svg"><path d="M11.571 4.714h1.715v5.143H11.57zm4.715 0H18v5.143h-1.714zM6 0L1.714 4.286v15.428h5.143V24l4.286-4.286h3.428L22.286 12V0zm14.571 11.143l-3.428 3.428h-3.429l-3 3v-3H6.857V1.714h13.714Z"/></svg>`;
        li.appendChild(lienTwitch);
      }
    });

    panel.appendChild(boutonFermer);
    panel.appendChild(header);
    panel.appendChild(liste);
    overlay.appendChild(panel);
    document.body.appendChild(overlay);
    modalEquipeActuelle = overlay;
    document.addEventListener("keydown", fermerModalSurEchap);
    boutonFermer.focus(); // pour la navigation clavier : le focus ne reste pas "perdu" derrière la modale
  }

  // Convertit une date du calendrier ("13 sept. 26") en vraie Date JS, à
  // 20h15 HEURE DE PARIS (heure officielle des journées de Ligue) —
  // nécessaire pour le décompte de la carte d'accueil et le rappel.
  // Correctif du 25/10/2026 : la Date était construite à 20h15 dans le
  // fuseau du NAVIGATEUR, donc décalée pour les joueurs hors de France.
  // Changements d'heure (été/hiver) gérés par Intl, sans table maison.
  const FORMAT_PARIS = new Intl.DateTimeFormat("en-US", {
    timeZone: "Europe/Paris",
    hourCycle: "h23",
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
  });

  // Décalage (en ms) de Paris par rapport à UTC à un instant donné.
  function decalageParisMs(instant) {
    const p = Object.fromEntries(FORMAT_PARIS.formatToParts(new Date(instant)).map((x) => [x.type, x.value]));
    const commeUtc = Date.UTC(+p.year, +p.month - 1, +p.day, +p.hour, +p.minute);
    return commeUtc - Math.floor(instant / 60000) * 60000;
  }

  // Date JS correspondant à annee/mois/jour heure:minute À PARIS. Deux
  // passes : le décalage dépend de l'instant (heure d'été ou d'hiver).
  function dateHeureParis(annee, mois, jour, heure, minute) {
    const naif = Date.UTC(annee, mois, jour, heure, minute);
    let instant = naif - decalageParisMs(naif);
    instant = naif - decalageParisMs(instant);
    return new Date(instant);
  }

  // Heure locale du joueur pour une Date, ou null si son fuseau donne la
  // même heure qu'à Paris (inutile de la répéter).
  function heureLocaleSiDifferente(date) {
    const locale = date.toLocaleTimeString(LANGUE === "en" ? "en-US" : "fr-FR", { hour: "2-digit", minute: "2-digit" });
    const paris = date.toLocaleTimeString(LANGUE === "en" ? "en-US" : "fr-FR", { hour: "2-digit", minute: "2-digit", timeZone: "Europe/Paris" });
    return locale === paris ? null : locale;
  }
  const MOIS_FR = {
    janv: 0, févr: 1, fev: 1, mars: 2, avr: 3, mai: 4, juin: 5,
    juil: 6, août: 7, aout: 7, sept: 8, oct: 9, nov: 10, déc: 11, dec: 11,
  };
  function parserDateCalendrier(dateStr) {
    const m = dateStr.match(/(\d+)\s+([a-zA-Zàâäéèêëïîôöùûüÿç]+)\.?\s+(\d+)/i);
    if (!m) return null;
    const jour = parseInt(m[1], 10);
    const moisAbbr = m[2].toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
    const mois = MOIS_FR[moisAbbr];
    if (mois === undefined) return null;
    const annee = 2000 + parseInt(m[3], 10);
    return dateHeureParis(annee, mois, jour, 20, 15);
  }

  // Cherche la prochaine ligne du calendrier qui contient au moins un
  // événement (ignore les semaines vides/pause) et n'est pas déjà
  // passée — utilisé par la carte d'accueil.
  function trouverProchaineJournee() {
    const maintenant = new Date();
    for (const ligne of CALENDRIER_S12) {
      const date = parserDateCalendrier(ligne.date);
      if (!date || date < maintenant) continue;
      if (ligne.jour0) return { date, libelle: t("jour0"), divisions: [] };
      const divisionsConcernees = [1, 2, 3, 4, 5].filter((d) => ligne[`div${d}`]);
      if (!divisionsConcernees.length) continue; // semaine sans événement, on cherche la suivante
      const libelles = [...new Set(divisionsConcernees.map((d) => traduireEvenementCalendrier(ligne[`div${d}`])))];
      return { date, libelle: libelles.join(" / "), divisions: divisionsConcernees };
    }
    return null;
  }

  function celluleCalendrier(row, cle) {
    if (row.jour0) {
      return `<td class="laligue-cal-jour0" colspan="5">${escapeHtml(t("jour0"))}</td>`;
    }
    const valeur = traduireEvenementCalendrier(row[cle]);
    return valeur
      ? `<td>${escapeHtml(valeur)}</td>`
      : `<td class="laligue-cal-vide">—</td>`;
  }

  function creerTableauCalendrier() {
    const lignesHtml = CALENDRIER_S12.map((row) => {
      const dateHtml = `<td>${escapeHtml(row.date)}</td>`;
      if (row.jour0) {
        return `<tr>${dateHtml}${celluleCalendrier(row, null)}</tr>`;
      }
      return `<tr>${dateHtml}${["div1", "div2", "div3", "div4", "div5"].map((cle) => celluleCalendrier(row, cle)).join("")}</tr>`;
    }).join("");

    const enTetesHtml = [1, 2, 3, 4, 5]
      .map((n) => {
        const c = COULEURS_DIVISIONS[n];
        return `<th style="background:${c.fond};color:${c.texte};">Division ${n}</th>`;
      })
      .join("");

    const wrap = document.createElement("div");
    wrap.className = "laligue-vue-calendrier-wrap";
    wrap.style.display = "none";
    wrap.innerHTML = `
      <table class="laligue-vue-calendrier">
        <thead><tr><th style="background:#1a1a1a;color:#999;">${escapeHtml(t("calendrierDate"))}</th>${enTetesHtml}</tr></thead>
        <tbody>${lignesHtml}</tbody>
      </table>
    `;
    return wrap;
  }

    function rendreZoneEquipes(filtre) {
      zoneEquipes.innerHTML = "";
      zoneEquipes.style.display = ""; // retire le style inline temporaire du squelette
      zoneEquipes.style.gridTemplateColumns = "";
      zoneEquipes.style.gap = "";

        if (!filtreNormalise) return true;

      });

        const vide = document.createElement("div");
        vide.className = "laligue-vue-vide";
        vide.textContent = t("aucunResultat");
        zoneEquipes.appendChild(vide);
        return;
      }

      const parNom = (a, b) => a.nom.localeCompare(b.nom);

      if (modeTriActuel === "alpha") {
        zoneEquipes.appendChild(creerGrille(filtrees.slice().sort(parNom)));
        return;
      }

      // Mode groupé par division : 1 → 5 puis les équipes sans division
      // assignée en dernier, chaque groupe trié alphabétiquement.
      const divisionsRepliees = obtenirDivisionsRepliees();
      const groupes = { 1: [], 2: [], 3: [], 4: [], 5: [], sans: [] };
      filtrees.forEach((equipe) => {
        (groupes[equipe.division] || groupes.sans).push(equipe);
      });

      [1, 2, 3, 4, 5, "sans"].forEach((cle) => {
        const equipesGroupe = groupes[cle];
        if (!equipesGroupe.length) return; // groupe vide après filtrage : pas d'en-tête inutile

        const estSansDivision = cle === "sans";
        const couleurs = estSansDivision ? { fond: "#333", texte: "#ccc" } : COULEURS_DIVISIONS[cle];
        const estRepliee = divisionsRepliees.has(String(cle));

        const entete = document.createElement("div");
        entete.className = "laligue-vue-division-entete";
        entete.style.background = couleurs.fond; /* repli si color-mix()/gradient non supporté */
        entete.style.background = `linear-gradient(135deg, ${couleurs.fond} 0%, color-mix(in srgb, ${couleurs.fond} 70%, black) 100%)`;
        entete.style.color = couleurs.texte;
        entete.innerHTML = `
          <span class="laligue-vue-division-entete-titre">
            <span class="laligue-vue-division-chevron${estRepliee ? " laligue-repliee" : ""}">▼</span>
            ${estSansDivision ? escapeHtml(t("sansDivision")) : `Division ${cle}`}
          </span>
          <span class="laligue-vue-division-actions">
            <span class="laligue-vue-division-compte">${equipesGroupe.length}</span>
          </span>
        `;

        // Liens vers le site officiel de l'organisation — seulement pour
        // les vraies divisions (le groupe "sans division" ne correspond à
        // aucune section là-bas). stopPropagation obligatoire : sans ça,
        // le clic remonterait à l'en-tête et replierait la division en
        // même temps qu'il ouvre le lien.
        if (!estSansDivision) {
          const actions = entete.querySelector(".laligue-vue-division-actions");
          const compte = actions.querySelector(".laligue-vue-division-compte");
          [
            { url: urlLigueStats(cle), libelle: t("lienStats"), titre: t("lienStatsTitre", cle), svg: SVG_STATS },
            { url: urlLigueClassement(cle), libelle: t("lienClassement"), titre: t("lienClassementTitre", cle), svg: SVG_CLASSEMENT },
          ].forEach(({ url, libelle, titre, svg }) => {
            const lien = document.createElement("a");
            lien.className = "laligue-vue-division-lien";
            lien.href = url;
            lien.target = "_blank";
            lien.rel = "noopener noreferrer";
            lien.title = titre;
            lien.innerHTML = `${svg}<span>${escapeHtml(libelle)}</span>`;
            lien.addEventListener("click", (e) => e.stopPropagation());
            actions.insertBefore(lien, compte);
          });
        }

        const grille = creerGrille(equipesGroupe.slice().sort(parNom));
        grille.style.display = estRepliee ? "none" : "grid";

        function basculerRepli() {
          const repliees = basculerDivisionRepliee(String(cle));
          const maintenantRepliee = repliees.has(String(cle));
          grille.style.display = maintenantRepliee ? "none" : "grid";
          entete.querySelector(".laligue-vue-division-chevron").classList.toggle("laligue-repliee", maintenantRepliee);
        }
        entete.addEventListener("click", basculerRepli);
        rendreAccessible(entete, basculerRepli, estSansDivision ? t("sansDivision") : `Division ${cle}`);

        const groupe = document.createElement("div");
        groupe.className = "laligue-vue-division-groupe";
        groupe.appendChild(entete);
        groupe.appendChild(grille);
        zoneEquipes.appendChild(groupe);
      });
    }

  function extraireIdentiteDeLigne(ligne) {
    // La ligne du SCORE PERSONNEL (mise en évidence, classe
    // "...selected__...") n'a souvent pas de lien /user/... — on retombe
    // sur le pseudo affiché, résolu par la même voie de repli que la
    // liste d'amis (voir resoudreEquipeJoueur), pour ne pas la perdre.
    const lien = ligne.querySelector('a[href^="/user/"]');
    const userId = lien ? (lien.getAttribute("href").match(/\/user\/([a-zA-Z0-9]+)/) || [])[1] || null : null;
    const nickSpan = ligne.querySelector('[class*="nick_nick__"]');
    const pseudo = nickSpan?.textContent.trim() || null;
    return { userId, pseudo };
  }

  function appliquerFiltreSurLignes() {
    const table = trouverTableResultats();
    if (!table) return;
    const filtreActif = filtreResultatsSelection.size > 0;
    table.querySelectorAll('[class*="coordinate-results_row__"]:not([class*="headerRow"])').forEach((ligne) => {
      const equipeId = ligne.dataset.laligueEquipeId;
      const visible = !filtreActif || (equipeId && filtreResultatsSelection.has(equipeId));
      ligne.style.display = visible ? "" : "none";
    });
    // Les séparateurs entre lignes restent visibles même quand la ligne
    // d'à côté est masquée, laissant des espaces vides disgracieux — on
    // les masque tous pendant qu'un filtre est actif, plus simple et
    // plus sûr que de calculer au cas par cas quelles paires de lignes
    // voisines sont encore visibles.
    table.querySelectorAll('[class*="coordinate-results_rowDivider__"]').forEach((diviseur) => {
      diviseur.style.display = filtreActif ? "none" : "";
    });
    filtrerClassementEquipes();
    synchroniserSwitchLlgg();
  }

  // Même sélection appliquée au classement des équipes (s'il est affiché) :
  // seules les équipes choisies restent visibles, avec leur rang calculé
  // sur TOUTES les équipes (le filtre n'est qu'un affichage).
  function filtrerClassementEquipes() {
    const panneau = document.getElementById("llgg-classement-equipes");
    if (!panneau) return;
    const filtreActif = filtreResultatsSelection.size > 0;
    panneau.querySelectorAll("[data-laligue-equipe-id]").forEach((el) => {
      el.style.display = !filtreActif || filtreResultatsSelection.has(el.dataset.laligueEquipeId) ? "" : "none";
    });
    const nonClassees = panneau.querySelector(".laligue-ce-non-classees");
    if (nonClassees) {
      // Virgules recalculées sur les seules équipes visibles.
      const visibles = [...nonClassees.querySelectorAll("[data-laligue-equipe-id]")].filter((el) => el.style.display !== "none");
      visibles.forEach((el, i) => (el.textContent = `${i ? ", " : ""}${el.dataset.texte}`));
      nonClassees.style.display = visibles.length ? "" : "none";
    }
  }

  // Point d'entrée unique pour changer la sélection : pastilles, "Toutes"

  function definirSelectionFiltreResultats(ids) {
    filtreResultatsSelection = new Set(ids);
    Object.entries(filtreResultatsChips).forEach(([id, { element }]) =>
      element.classList.toggle("laligue-filtre-actif", filtreResultatsSelection.has(id))
    );
    if (filtreResultatsChipTous) {
      filtreResultatsChipTous.classList.toggle("laligue-filtre-actif", filtreResultatsSelection.size === 0);
    }
    appliquerFiltreSurLignes();
  }

  }

  // ---------------------------------------------------------------
  // Classement provisoire des équipes (page de résultats d'un challenge)
  // ---------------------------------------------------------------
  // Règles de la Ligue (maps coop, laliguegeoguessr.fr/fr/ligue/rules.html) :
  // - une équipe = la somme de 3 joueurs. Ici : ses 3 joueurs au meilleur
  //   score total (une équipe en a souvent plus sur un challenge ouvert) ;
  //   moins de 3 joueurs = non classée (forfait dans le règlement) ;
  // - sur chaque manche, et sur le score total compté comme 6e manche,
  //   une équipe gagne autant de points qu'il y a d'équipes ayant fait
  //   aussi bien ou moins bien (les ex-æquo prennent donc la place haute) ;
  // - points de map : même principe sur la somme des points de rounds.
  // - scoring par seuil sur les scores de manche et le total (voir
  //   pointsAvecSeuil).
  // Pas repris : le multiplicateur x3 des maps coop (il ne change pas
  // l'ordre sur une seule map).
  // Données : l'API des résultats du challenge (celle de la page), avec
  // le détail des manches de chaque joueur, par pages, une à la fois.
  // Seuls les joueurs reconnus PAR ID comptent (pas de déduction par
  // pseudo pour un classement).
  async function recupererResultatsChallenge(jeton) {
    const joueurs = [];
    let pagination = "";
    for (let page = 0; page < 40; page++) {
      if (page) await pause(80);
      const url =
        `${location.origin}/api/v3/results/highscores/${encodeURIComponent(jeton)}?friends=false&limit=50&minRounds=5` +
        (pagination ? `&paginationToken=${encodeURIComponent(pagination)}` : "");
      const reponse = await fetch(url, { credentials: "include" });
      if (!reponse.ok) throw new Error(`HTTP ${reponse.status}`);
      const json = await reponse.json();
      const items = Array.isArray(json?.items) ? json.items : [];
      items.forEach((it) => {
        const joueur = it.game?.player || {};
        const manches = (Array.isArray(joueur.guesses) ? joueur.guesses : []).map((g) => Number(g.roundScoreInPoints) || 0);
        joueurs.push({
          userId: String(it.userId || joueur.id || ""),
          pseudo: String(it.playerName || joueur.nick || ""),
          total: Number(it.totalScore ?? joueur.totalScoreInPoints ?? joueur.totalScore?.amount) || 0,
          manches,
        });
      });
      pagination = json?.paginationToken || "";
      if (!pagination || !items.length) break;
    }
    return joueurs;
  }

  function calculerClassementEquipes(joueurs, donnees) {
    const parEquipe = {};
    joueurs.forEach((j) => {
      const entree = donnees.joueurs[j.userId];
      if (!entree || !donnees.equipes[entree.equipeId] || j.manches.length < 5) return;
      (parEquipe[entree.equipeId] = parEquipe[entree.equipeId] || []).push(j);
    });
    const classees = [];
    const nonClassees = [];
    Object.entries(parEquipe).forEach(([id, liste]) => {
      const equipe = donnees.equipes[id];
      if (liste.length < 3) {
        nonClassees.push({ equipe, nombre: liste.length });
        return;
      }
      const trio = liste.sort((a, b) => b.total - a.total).slice(0, 3);
      // 5 manches + le total en 6e manche
      const scores = [0, 1, 2, 3, 4].map((i) => trio.reduce((somme, j) => somme + (j.manches[i] || 0), 0));
      scores.push(trio.reduce((somme, j) => somme + j.total, 0));
      classees.push({ equipe, trio, scores });
    });
    const pointsSelon = (valeur, toutes) => toutes.filter((v) => v <= valeur).length;
    // Scoring par seuil (règlement) : "10 % du score manquant sur le score
    // total du round, amenant à des égalités avec les joueurs qui sont
    // dans votre seuil". Score maximal d'une équipe de 3 : 15 000 par
    // manche, 75 000 sur le total. Ex. : 13 000 sur une manche → 2 000
    // manquants → seuil de 200 → toute équipe jusqu'à 13 200 compte comme
    // ayant fait "aussi bien".
    const pointsAvecSeuil = (valeur, toutes, maximum) => pointsSelon(valeur + 0.1 * Math.max(0, maximum - valeur), toutes);
    for (let m = 0; m < 6; m++) {
      const colonne = classees.map((c) => c.scores[m]);
      const maximum = m < 5 ? 15000 : 75000;
      classees.forEach((c) => {
        (c.points = c.points || [])[m] = pointsAvecSeuil(c.scores[m], colonne, maximum);
      });
    }
    classees.forEach((c) => (c.ptsRounds = c.points.reduce((a, b) => a + b, 0)));
    const totaux = classees.map((c) => c.ptsRounds);
    classees.forEach((c) => (c.ptsMap = pointsSelon(c.ptsRounds, totaux)));
    classees.sort((a, b) => b.ptsMap - a.ptsMap || b.scores[5] - a.scores[5]);
    // Rang affiché : ex-æquo au même rang (place la plus haute)
    classees.forEach((c, i) => (c.rang = i && classees[i - 1].ptsMap === c.ptsMap ? classees[i - 1].rang : i + 1));
    nonClassees.sort((a, b) => b.nombre - a.nombre);
    return { classees, nonClassees };
  }

  function construirePanneauClassement({ classees, nonClassees }) {
    const panneau = document.createElement("div");
    panneau.className = "laligue-classement-equipes";
    if (!classees.length) {
      panneau.textContent = t("classementEquipesVide");
    } else {
      const table = document.createElement("table");
      const entete = document.createElement("tr");
      ["#", t("colEquipe"), "R1", "R2", "R3", "R4", "R5", t("colTotalManche"), t("colPtsRounds"), t("colPtsMap")].forEach((texte, i) => {
        const th = document.createElement("th");
        th.textContent = texte;
        if (i === 1) th.className = "laligue-ce-gauche";
        entete.appendChild(th);
      });
      const thead = document.createElement("thead");
      thead.appendChild(entete);
      table.appendChild(thead);
      const tbody = document.createElement("tbody");
      classees.forEach((c) => {
        const tr = document.createElement("tr");
        tr.dataset.laligueEquipeId = c.equipe.tag.toLowerCase(); // même identifiant que les pastilles
        const cellule = (classe, texte) => {
          const td = document.createElement("td");
          if (classe) td.className = classe;
          if (texte !== undefined) td.textContent = texte;
          tr.appendChild(td);
          return td;
        };
        cellule("laligue-ce-rang", String(c.rang));
        const tdEquipe = cellule("laligue-ce-gauche");
        const ligneEquipe = document.createElement("div");
        ligneEquipe.className = "laligue-ce-equipe";
        ligneEquipe.appendChild(creerBadge(c.equipe));
        ligneEquipe.appendChild(document.createTextNode(c.equipe.nom));
        tdEquipe.appendChild(ligneEquipe);
        const joueursEl = document.createElement("div");
        joueursEl.className = "laligue-ce-joueurs";
        joueursEl.textContent = c.trio.map((j) => j.pseudo).join(" · ");
        tdEquipe.appendChild(joueursEl);
        // Place sur la manche (ex-æquo par seuil = même place) puis les
        // points entre parenthèses : place = nombre d'équipes − points + 1.
        c.points.forEach((pts, m) => {
          const place = classees.length - pts + 1;
          const td = cellule("laligue-ce-pts", `${t("classementPoints", place, "")} (${pts})`);
          const score = document.createElement("span");
          score.className = "laligue-ce-score";
          score.textContent = c.scores[m].toLocaleString(LANGUE === "en" ? "en-US" : "fr-FR");
          td.appendChild(score);
        });
        cellule("laligue-ce-pts", String(c.ptsRounds));
        cellule("laligue-ce-map", String(c.ptsMap));
        tbody.appendChild(tr);
      });
      table.appendChild(tbody);
      panneau.appendChild(table);
    }
    if (nonClassees.length) {
      const bloc = document.createElement("div");
      bloc.className = "laligue-ce-non-classees";
      bloc.textContent = `${t("classementEquipesNonClassees")} `;
      nonClassees.forEach((n, i) => {
        const item = document.createElement("span");
        item.dataset.laligueEquipeId = n.equipe.tag.toLowerCase();
        item.dataset.texte = `${n.equipe.tag} (${n.nombre})`;
        item.textContent = `${i ? ", " : ""}${item.dataset.texte}`;
        bloc.appendChild(item);
      });
      panneau.appendChild(bloc);
    }
    const note = document.createElement("div");
    note.className = "laligue-ce-note";
    note.textContent = t("classementEquipesNote");
    panneau.appendChild(note);
    const mode = document.createElement("div");
    mode.className = "laligue-ce-note";
    mode.textContent = t("classementEquipesMode");
    panneau.appendChild(mode);
    const officiel = document.createElement("div");
    officiel.className = "laligue-ce-note";
    officiel.textContent = t("classementEquipesOfficiel");
    panneau.appendChild(officiel);
    return panneau;
  }

  let classementEquipesEnCours = false;

  async function basculerClassementEquipes(barre, bouton) {
    const existant = document.getElementById("llgg-classement-equipes");
    if (existant) {
      existant.remove();
      bouton.textContent = t("classementEquipes");
      return;
    }
    if (classementEquipesEnCours) return;
    const jeton = (location.pathname.match(/^\/results\/([A-Za-z0-9]+)/) || [])[1];
    if (!jeton) return;
    classementEquipesEnCours = true;
    const panneau = document.createElement("div");
    panneau.id = "llgg-classement-equipes";
    panneau.className = "laligue-classement-equipes";
    panneau.textContent = t("classementEquipesCalcul");
    barre.insertAdjacentElement("afterend", panneau);
    bouton.textContent = t("classementEquipesMasquer");
    try {
      const [donnees, joueurs] = await Promise.all([obtenirDonneesLigue(), recupererResultatsChallenge(jeton)]);
      if (!panneau.isConnected) return; // masqué entre-temps
      const classement = calculerClassementEquipes(joueurs, donnees);
      // Pastille de filtre pour chaque équipe du classement, même si ses
      // lignes ne sont pas (encore) chargées dans le tableau du jeu.
      [...classement.classees, ...classement.nonClassees].forEach(({ equipe }) => ajouterChipEquipeSiNouvelle(equipe));
      const resultat = construirePanneauClassement(classement);
      resultat.id = panneau.id;
      panneau.replaceWith(resultat);
      filtrerClassementEquipes();
    } catch (e) {
      console.warn("[LLGG] Classement des équipes impossible :", e);
      panneau.textContent = t("classementEquipesErreur");
    } finally {
      classementEquipesEnCours = false;
    }
  }

  let switchLlggEquipeId = null;

  function trouverMonEquipeResultats() {
    // Même règle que partout : l'ID du joueur connecté fait foi (voir
    // trouverMonEquipe), jamais l'équipe déduite d'une ligne affichée.
    return donneesLigueEnMemoire ? trouverMonEquipe(donneesLigueEnMemoire) : null;

  function synchroniserSwitchLlgg() {
    const notre = document.querySelector(".laligue-switch-llgg");
    if (!notre) return;
    const racine = notre.parentElement;
    const actif = !!switchLlggEquipeId && filtreResultatsSelection.size === 1 && filtreResultatsSelection.has(switchLlggEquipeId);
    racine.dataset.laligueModeLlgg = actif ? "1" : "0";
    notre.querySelector("label")?.setAttribute("aria-selected", actif ? "true" : "false");
    const fond = notre.querySelector(".laligue-switch-llgg-bg");
    if (fond) fond.className = `${actif ? notre.dataset.clsShow : notre.dataset.clsHide} laligue-switch-llgg-bg`;
  }

  function tenterInjectionSwitchLlgg() {
    const racine = document.querySelector('[class*="results_switch__"] [class*="switch_switch__"]');
    if (!racine) return;
    if (!estActif() || !estFiltreResultatsActif() || !donneesLigueEnMemoire || !filtreResultatsBarre) {
      racine.querySelector(".laligue-switch-llgg")?.remove();
      return;
    }
    if (racine.querySelector(".laligue-switch-llgg")) {
      synchroniserSwitchLlgg();
      return;
    }

    const equipe = trouverMonEquipeResultats();
    if (!equipe) return; // joueur non identifié ou hors Ligue
    const items = [...racine.querySelectorAll('[class*="switch_switchItem__"]')];
    const fondShow = racine.querySelector('[class*="switch_background__"][class*="switch_show"]');
    const fondHide = racine.querySelector('[class*="switch_background__"][class*="switch_hide"]');
    const labelModele = racine.querySelector('[class*="switch_label__"]');
    if (!items.length || !fondShow || !fondHide || !labelModele) return;

    switchLlggEquipeId = equipe.tag.toLowerCase();
    const item = document.createElement("div");
    item.className = `${items[0].className} laligue-switch-llgg`;
    item.dataset.clsShow = fondShow.className;
    item.dataset.clsHide = fondHide.className;
    item.title = t("ongletLlggTitre", equipe.nom);
    const fond = document.createElement("div");
    fond.className = `${fondHide.className} laligue-switch-llgg-bg`; // classe de repérage dès la création, voir synchroniserSwitchLlgg
    const label = document.createElement("label");
    label.className = labelModele.className;
    const etiquette = document.createElement("span");
    etiquette.className = "laligue-onglet-llgg-tag";
    etiquette.textContent = equipe.tag;
    etiquette.style.background = equipe.couleurFond;
    etiquette.style.color = couleurTexteSure(equipe.couleurFond, equipe.couleurTexte);
    label.appendChild(etiquette);
    item.appendChild(fond);
    item.appendChild(label);
    racine.appendChild(item);

    // Couleur "non sélectionné" relevée sur un label natif, pour atténuer
    // l'item natif qui reste sélectionné côté React.
    const labelInactif = racine.querySelector('[class*="switch_label__"][aria-selected="false"]');
    if (labelInactif) racine.style.setProperty("--llgg-tab-inactif", getComputedStyle(labelInactif).color);

    const activer = () => {
      // L'item "All" en premier : c'est le seul qui contient forcément
      // toute l'équipe. Repéré par position (le libellé dépend de la
      // langue de GeoGuessr).
      const labelTous = items[0].querySelector('[class*="switch_label__"]');
      if (labelTous && labelTous.getAttribute("aria-selected") !== "true") labelTous.click();
      definirSelectionFiltreResultats([switchLlggEquipeId]);
    };
    item.addEventListener("click", (e) => {
      e.preventDefault();
      e.stopPropagation();
      activer();
    });
    rendreAccessible(item, activer, t("ongletLlggTitre", equipe.nom));

    // Choisir un item natif ensuite = sortir de notre vue filtrée. Garde
    // anti-doublon : l'item peut être recréé sur le même conteneur (réglage
    // désactivé puis réactivé sans recharger), l'écouteur ne doit pas
    // s'empiler.
    if (!racine.dataset.laligueEcoute) {
      racine.dataset.laligueEcoute = "1";
      racine.addEventListener("pointerdown", (e) => {
        const cible = e.target instanceof Element && e.target.closest('[class*="switch_switchItem__"]');
        if (cible && !cible.classList.contains("laligue-switch-llgg") && racine.dataset.laligueModeLlgg === "1") {
          definirSelectionFiltreResultats([]);
        }
      }, true);
    }

    synchroniserSwitchLlgg();
  }

  function ajouterChipEquipeSiNouvelle(equipe) {
    const equipeId = equipe.tag.toLowerCase();
    if (filtreResultatsChips[equipeId]) return;
    const chip = document.createElement("span");
    chip.className = "laligue-filtre-chip";
    chip.textContent = equipe.tag;
    chip.style.background = equipe.couleurFond;
    chip.style.color = couleurTexteSure(equipe.couleurFond, equipe.couleurTexte);
    chip.addEventListener("click", () => basculerChipEquipe(equipeId));
    rendreAccessible(chip, () => basculerChipEquipe(equipeId), equipe.nom);
    filtreResultatsChips[equipeId] = { element: chip, nom: equipe.nom };
    filtreResultatsBarre.insertBefore(chip, ancrePastillesResultats());
    trierChipsEquipes();
  }

  // Charge automatiquement le reste de la liste (au-delà des 25 premiers
  // résultats). Ancre confirmée par inspection DOM (16/08/2026) :
  // GeoGuessr utilise un vrai bouton "voir plus" (dans un conteneur
  // `results_center__`), pas un scroll infini. On clique dessus en
  // boucle jusqu'à ce qu'il disparaisse (plus de résultats à charger).
  function trouverBoutonVoirPlus() {
    return document.querySelector('[class*="results_center__"] button');
  }

  async function chargerTousLesResultats() {
    if (!filtreResultatsBoutonCharger) return;

    filtreResultatsBoutonCharger.disabled = true;
    filtreResultatsBoutonCharger.textContent = t("chargement");

    for (let i = 0; i < 40; i++) {
      const boutonVoirPlus = trouverBoutonVoirPlus();
      if (!boutonVoirPlus) break; // plus de bouton = tout est chargé
      boutonVoirPlus.click();
      await new Promise((r) => setTimeout(r, 600)); // laisse le temps au nouveau lot de charger et de se rendre
    }

    filtreResultatsBoutonCharger.disabled = false;
    filtreResultatsBoutonCharger.textContent = t("chargerTout");
    tenterInjectionFiltreResultats(); // traite les lignes fraîchement apparues
  }

  let filtreResultatsEnCours = false;

  async function tenterInjectionFiltreResultats() {
    if (!estActif() || !estFiltreResultatsActif()) return;
    if (filtreResultatsEnCours) return;

    const table = trouverTableResultats();
    if (!table) return;

    if (!filtreResultatsBarre || !filtreResultatsBarre.isConnected) {
      // Nouvelle page de résultats (ou premier passage) : l'ancienne
      // barre a été détruite par React — on repart entièrement à zéro
      // (sélection + pastilles déjà connues), sinon les pastilles des
      // équipes déjà vues sur un précédent challenge ne seraient jamais
      // recréées dans la nouvelle barre.
      filtreResultatsSelection = new Set();
      filtreResultatsChips = {};
      filtreResultatsBarre = creerBarreFiltreResultats();
      table.insertAdjacentElement("beforebegin", filtreResultatsBarre);
      // Chargement automatique de tous les résultats dès l'arrivée sur
      // la page, plus besoin de cliquer "Charger tous les résultats" —
      // le bouton reste utile en repli manuel si de nouveaux résultats
      // tombent entre-temps (parties encore en cours par exemple).
      chargerTousLesResultats();
    }

    // Pas de filtre ":not([data-laligue-scanne])" ici, volontairement :
    // contrairement à l'injection des badges (qui créerait des doublons
    // visuels si ré-exécutée), recalculer l'équipe d'une ligne déjà vue
    // ne coûte rien (pas d'appel réseau) — et ça permet de rattraper les
    // lignes dont le lien /user/... n'était pas encore rendu par React
    // lors d'un premier passage trop précoce.
    const lignes = table.querySelectorAll('[class*="coordinate-results_row__"]:not([class*="headerRow"])');
    if (!lignes.length) return;

    filtreResultatsEnCours = true;
    try {
      const donnees = await obtenirDonneesLigue();
      const label = document.getElementById("llgg-filtre-label");
      if (label) label.textContent = t("filtrerParEquipe");
      document.getElementById("llgg-filtre-squelettes")?.remove();
      lignes.forEach((ligne) => {
        const { userId, pseudo } = extraireIdentiteDeLigne(ligne);
        const equipe = resoudreEquipeJoueur(donnees, { userId, pseudo });
        if (equipe && userId && pseudo) enregistrerPseudoAppris(pseudo, userId);
        ligne.dataset.laligueEquipeId = equipe ? equipe.tag.toLowerCase() : "";
        if (equipe) ajouterChipEquipeSiNouvelle(equipe);
      });
      appliquerFiltreSurLignes();
    } finally {
      filtreResultatsEnCours = false;
    }
  }

      .llgg-tuile-notif svg { width: 15px; height: 15px; display: block; }
      .llgg-tuile-notif:hover { color: #fff; background: rgba(255, 255, 255, 0.08); }
      .llgg-tuile-notif.laligue-notif-active { color: var(--llgg-accent); }
    `;
    document.head.appendChild(style);

  function formaterDecompte(msRestant) {
    if (msRestant <= 0) return "—";
    const totalSecondes = Math.floor(msRestant / 1000);
    const jours = Math.floor(totalSecondes / 86400);
    const heures = Math.floor((totalSecondes % 86400) / 3600);
    const minutes = Math.floor((totalSecondes % 3600) / 60);
    const secondes = totalSecondes % 60;
    const pad = (n) => String(n).padStart(2, "0");
    // Au-delà d'un jour, les secondes qui défilent n'apportent rien et
    // polluent la lecture — on simplifie à "Xj" (ou "Xj HHh" si on
    // préfère un peu plus de précision), le détail HH:MM:SS n'a
    // vraiment de sens que dans les dernières 24h.
    if (jours > 0) return `${jours}${t("accueilJourAbrev")} ${pad(heures)}h`;
    return `${pad(heures)}:${pad(minutes)}:${pad(secondes)}`;
  }

  // Applique la position sauvegardée (si le joueur a déjà déplacé la
  // tuile) — sinon on laisse le CSS gérer la position par défaut
  // (left/bottom fixes), donc rien à faire ici tant que rien n'a jamais
  // été sauvegardé.
  // Ramène l'élément dans les limites visibles de la fenêtre s'il en
  // déborde (ex: position sauvegardée sur un grand écran, rouverte sur
  // un écran plus petit — la tuile pourrait sinon se retrouver
  // totalement hors champ, invisible et inaccessible). Renvoie true si
  // une correction a été appliquée.
  function clamperElementDansViewport(el) {
    const marge = 4;
    const rect = el.getBoundingClientRect();
    if (rect.width === 0 && rect.height === 0) return false; // pas encore rendu, rien à corriger
    const maxLeft = Math.max(window.innerWidth - rect.width - marge, marge);
    const maxTop = Math.max(window.innerHeight - rect.height - marge, marge);
    let left = Math.min(Math.max(rect.left, marge), maxLeft);
    let top = Math.min(Math.max(rect.top, marge), maxTop);
    const modifie = left !== rect.left || top !== rect.top;
    if (modifie) {
      el.style.left = `${left}px`;
      el.style.top = `${top}px`;
      el.style.bottom = "auto";
    }
    return modifie;
  }

  function appliquerPositionTuileSauvegardee(tuile) {
    try {
      const pos = JSON.parse(localStorage.getItem(CLE_TUILE_POSITION));
      if (pos && typeof pos.left === "number" && typeof pos.top === "number") {
        tuile.style.left = `${pos.left}px`;
        tuile.style.top = `${pos.top}px`;
        tuile.style.bottom = "auto"; // sinon top ET bottom fixés en même temps déformeraient l'élément
      }
    } catch (e) {
      // pas grave, position par défaut du CSS
    }
  }

  // Glisser-déposer classique à la souris. Un simple clic (sans
  // déplacement) doit continuer à ouvrir la vue Ligue normalement — on
  // ne bascule en mode "glissé" qu'au-delà d'un petit seuil de
  // mouvement, et on bloque alors le clic qui suivrait le relâchement.
  function rendreTuileDeplacable(tuile) {
    const SEUIL_PX = 4;
    let enGlissement = false;
    let venaitDeGlisser = false;
    let decalageX = 0;
    let decalageY = 0;
    let departX = 0;
    let departY = 0;

    tuile.addEventListener("mousedown", (e) => {
      if (e.target.closest("button")) return; // laisse les boutons internes (fermer, cloche) fonctionner normalement
      enGlissement = true;
      venaitDeGlisser = false;
      departX = e.clientX;
      departY = e.clientY;
      const rect = tuile.getBoundingClientRect();
      decalageX = e.clientX - rect.left;
      decalageY = e.clientY - rect.top;
      e.preventDefault(); // évite la sélection de texte pendant le glissement
    });

    document.addEventListener("mousemove", (e) => {
      if (!enGlissement) return;
      if (!venaitDeGlisser && Math.hypot(e.clientX - departX, e.clientY - departY) < SEUIL_PX) return;
      venaitDeGlisser = true;
      tuile.style.cursor = "grabbing";
      tuile.style.left = `${e.clientX - decalageX}px`;
      tuile.style.top = `${e.clientY - decalageY}px`;
      tuile.style.bottom = "auto";
    });

    document.addEventListener("mouseup", () => {
      if (!enGlissement) return;
      enGlissement = false;
      tuile.style.cursor = "";
      if (venaitDeGlisser) {
        const rect = tuile.getBoundingClientRect();
        try {
          localStorage.setItem(CLE_TUILE_POSITION, JSON.stringify({ left: rect.left, top: rect.top }));
        } catch (e) {
          // pas grave si indisponible
        }
      }
    });

    // Capturé dans la fermeture : le handler de clic (ouverture de la
    // vue Ligue) le consulte pour savoir s'il doit s'annuler.
    return () => venaitDeGlisser;
  }

  // Regroupe une rafale de mutations en un seul traitement par frame
  // (même principe que dans Cartometa — évite de retraiter le DOM à
  // chaque micro-mutation pendant une animation ou un scroll).
  function grouperParFrame(fn) {
    let planifie = false;
    return function (...args) {
      if (planifie) return;
      planifie = true;
      requestAnimationFrame(() => {
        planifie = false;
        fn(...args);
      });
    };
  }

  function extraireUserId(wrapper) {
    const lien =
      wrapper.closest('a[href^="/user/"]') || wrapper.querySelector('a[href^="/user/"]');
    if (!lien) return null;
    const match = lien.getAttribute("href").match(/\/user\/([a-zA-Z0-9]+)/);
    return match ? match[1] : null;
  }

  // Dans certains menus/classements, GeoGuessr affiche le joueur connecté
  // sous un libellé générique ("You", "Moi"…) au lieu de son pseudo, et
  // sans lien /user/ : ni l'ID ni le pseudo ne permettaient de le
  // résoudre (bug du 19/10/2026). Libellés des langues du site ; comparés
  // sans casse, et seulement en l'absence de lien /user/.
  const LIBELLES_MOI = new Set([
    "you", "moi", "vous", "du", "tú", "tu", "você", "jij", "ty", "вы", "ты",
  ]); // volontairement restreint : un vrai pseudo identique recevrait notre badge
  function estLibelleMoi(texte) {
    return !!texte && LIBELLES_MOI.has(texte.toLocaleLowerCase());
  }

  // ID du joueur connecté : mémorisé (CLE_MON_ID, capturé sur /me/profile)
  // ou demandé UNE fois par session à l'API GeoGuessr (même origine,
  // cookies de session). Promesse partagée : un seul appel même si
  // plusieurs "You" sont traités en même temps, et pas de nouvel essai
  // en boucle en cas d'échec.
  let promesseMonId = null;
  function obtenirMonId() {
    const memorise = localStorage.getItem(CLE_MON_ID);
    if (memorise) return Promise.resolve(memorise);
    if (!promesseMonId) {
      // URL absolue : dans Firefox, le fetch d'un script de contenu ne
      // résout pas toujours une URL relative par rapport à la page.
      promesseMonId = fetch(`${location.origin}/api/v3/profiles`, { credentials: "include" })
        .then((r) => (r.ok ? r.json() : null))
        .then((json) => {
          const id = json?.user?.id || json?.user?.userId || null;
          if (id && /^[0-9a-f]{24}$/.test(id)) {
            try {
              localStorage.setItem(CLE_MON_ID, id);
            } catch (e) {
              // pas grave
            }
            return id;
          }
          return null;
        })
        .catch(() => null);
    }
    return promesseMonId;
  }

  async function traiterWrapper(wrapper) {
    // estActif() est vérifié une fois par scan dans scannerEtInjecter,
    // seul appelant.
    const nickSpan = wrapper.querySelector('[class*="nick_nick__"]');
    let pseudo = nickSpan?.textContent.trim();
    let userId = extraireUserId(wrapper);
    if (!userId && estLibelleMoi(pseudo)) {
      userId = await obtenirMonId();
      if (!userId) return; // inconnu : on ne tente surtout pas "You" comme pseudo
      pseudo = null; // ni apprentissage ni vérification sur un libellé générique
    }
    if (!userId && !pseudo) return; // rien d'exploitable encore, on retentera au prochain scan (pas de marquage)

    // Certains tableaux (classement Duels notamment) semblent recycler
    // les mêmes éléments DOM pour afficher des joueurs différents au fil
    // du défilement/live, sans jamais les retirer du DOM — si le joueur
    // affiché a changé depuis notre dernier passage sur CET élément
    // précis, on retraite de zéro plutôt que de faire confiance à un
    // marquage périmé (et on retire l'éventuel badge de l'ancien
    // joueur, sinon il resterait affiché sur le nouveau par erreur).
    const identiteActuelle = userId || pseudo;
    if (wrapper.dataset.laligueTraite) {
      if (wrapper.dataset.laligueIdentite === identiteActuelle) return; // même joueur, rien à refaire
      wrapper.querySelector(":scope > .laligue-badge")?.remove();
    }
    wrapper.dataset.laligueTraite = "1";
    wrapper.dataset.laligueIdentite = identiteActuelle;

    const donnees = await obtenirDonneesLigue();
    const equipe = resoudreEquipeJoueur(donnees, { userId, pseudo });
    if (!equipe) {
      logDebug(`Pas de correspondance pour userId="${userId}" pseudo="${pseudo}"`);
      return; // joueur non inscrit à la Ligue, on ne touche à rien
    }

    // Résolution réussie via un ID fiable : on retient ce pseudo pour
    // ce joueur, réutilisable en repli dans les contextes sans ID (voir
    // enregistrerPseudoAppris plus haut).
    if (userId && pseudo) enregistrerPseudoAppris(pseudo, userId);

    // Tag posé par déduction (pseudo) : vérification en tâche de fond,
    // sans bloquer l'affichage — voir verifierCorrespondancePseudo.
    if (equipe.fiabilite === "pseudo") verifierCorrespondancePseudo(pseudo, equipe.idCandidat);

    // Garde-fou INCONDITIONNEL contre les doublons : peu importe ce que
    // dit le marquage "traité" (qui peut être perturbé par une remise à
    // zéro concurrente déclenchée par l'apprentissage d'un AUTRE joueur
    // pendant qu'on est encore en plein traitement de celui-ci), on
    // retire systématiquement tout badge (et son icône de signalement,
    // désormais séparée) déjà présents avant d'en reposer — jamais plus
    // d'un, sans exception.
    wrapper.querySelectorAll(":scope > .laligue-badge").forEach((b) => b.remove());
    wrapper.querySelectorAll(":scope > .laligue-badge-signaler").forEach((s) => s.remove());

    const badge = creerBadge(equipe, pseudo);
    // "Avant le pseudo" : inséré juste devant le span du nick, à
    // l'intérieur du wrapper, pour hériter naturellement de son
    // alignement flex (même registre visuel que le clubTag natif de
    // GeoGuessr, qui suit le même schéma). On insère devant l'enfant
    // DIRECT qui contient le pseudo, et non devant le pseudo lui-même :
    // s'il est un jour enveloppé (ex. dans un lien), insertBefore lèverait
    // une erreur et aucun badge n'apparaîtrait sur tout l'écran.
    const ancreBadge = nickSpan ? [...wrapper.children].find((enfant) => enfant.contains(nickSpan)) : null;
    wrapper.insertBefore(badge, ancreBadge || wrapper.firstChild);

    // Icône de signalement tout à la fin de la ligne (après le drapeau
    // et les éventuels badges natifs du joueur), pas juste après le
    // pseudo — moins intrusif visuellement.
    if (equipe.fiabilite === "pseudo") {
      wrapper.appendChild(creerIconeSignalement(pseudo));
    }
  }

  let demandeMonIdLancee = false;

  function trouverMonEquipe(donnees) {
    const monId = localStorage.getItem(CLE_MON_ID);
    if (!monId) {

  function activerOngletNatif(onglet) {
    // Composant d'onglets maison de GeoGuessr (pas Radix) : on ignore sur
    // quel événement il réagit, donc on rejoue la séquence complète d'un
    // vrai clic — pointer*, mouse*, click — puis le focus. Correctif du
    // 19/10/2026 : sans les pointer*, la bascule sur "All" échouait dans
    // le défi du jour. Idempotent si l'onglet est déjà sélectionné.
    const opts = { bubbles: true, cancelable: true, button: 0, buttons: 1, pointerId: 1, isPrimary: true, pointerType: "mouse" };
    onglet.dispatchEvent(new PointerEvent("pointerdown", opts));
    onglet.dispatchEvent(new MouseEvent("mousedown", opts));
    onglet.dispatchEvent(new PointerEvent("pointerup", { ...opts, buttons: 0 }));
    onglet.dispatchEvent(new MouseEvent("mouseup", { ...opts, buttons: 0 }));
    onglet.click();
    onglet.focus({ preventScroll: true });
  }

  // Intention de l'utilisateur, gardée HORS du DOM : React peut remonter
  // toute la carte de classement (changement d'onglet, nouveau round),

  // Applique l'état visuel sans toucher à l'intention.
  function afficherModeLlgg(racineOnglets, actif) {
    // Écritures conditionnelles : appelée à chaque scan, et chaque
    // écriture d'attribut relancerait le MutationObserver.
    const valeur = actif ? "1" : "0";
    if (racineOnglets.dataset.laligueModeLlgg !== valeur) racineOnglets.dataset.laligueModeLlgg = valeur;
    const notre = racineOnglets.querySelector(".laligue-onglet-llgg");
    if (notre) {
      const etat = actif ? "active" : "inactive";
      if (notre.dataset.state !== etat) notre.dataset.state = etat;
      if (notre.getAttribute("aria-selected") !== String(actif)) notre.setAttribute("aria-selected", String(actif));
    }
  }

  // Deux contextes connus (19/10/2026) :
  // - entre les rounds / fin de partie : barre round-leaderboard_toolbar__
  //   au-dessus d'un tableau round-leaderboard_row__ ;
  // - récap du défi du jour : mêmes onglets natifs (+ "My game" et pays),
  //   mais structure de liste non inspectée — d'où une détection
  //   générique : on remonte depuis les onglets jusqu'au premier ancêtre
  //   qui contient des pseudos, et chaque pseudo donne sa ligne.
  const SEL_NICK = '[class*="player-nick_wrapper"]';

  function trouverCarteClassement(racineOnglets) {
    const toolbar = racineOnglets.closest('[class*="round-leaderboard_toolbar__"]');
    if (toolbar) return toolbar.parentElement;
    let el = racineOnglets.parentElement;
    for (let i = 0; el && i < 5; i++, el = el.parentElement) {
      if ([...el.querySelectorAll(SEL_NICK)].some((w) => !racineOnglets.contains(w))) return el;
    }
    return null;
  }

  function lignesClassement(carte, racineOnglets) {
    const tbodyRound = carte.querySelector('[class*="round-leaderboard_table__"] tbody');
    if (tbodyRound) return [...tbodyRound.querySelectorAll('tr[class*="round-leaderboard_row__"]')];
    const lignes = new Set();
    carte.querySelectorAll(SEL_NICK).forEach((w) => {
      if (racineOnglets.contains(w)) return;
      // parentElement : le wrapper lui-même porte parfois un nom en "Row__".
      const ligne = w.closest("tr") || w.parentElement?.closest('li, [class*="_row__"], [class*="Row__"]') || w;
      if (carte.contains(ligne) && ligne !== carte) lignes.add(ligne);
    });
    return [...lignes];
  }

  // carte : optionnelle, pour éviter de la rechercher deux fois par scan.
  function appliquerFiltreOngletLlgg(racineOnglets, equipe, carte = trouverCarteClassement(racineOnglets)) {
    if (!carte) return;
    const actif = racineOnglets.dataset.laligueModeLlgg === "1";
    // Ajouts de coéquipiers : best-effort, une erreur ici ne doit JAMAIS
    // empêcher le filtrage des lignes natives (bug du 22/10/2026).
    // Pas d'équivalent entre les rounds : le classement d'un round est
    // plafonné à 5 joueurs côté serveur, et les résultats complets ne sont
    // servis qu'aux joueurs ayant TERMINÉ le challenge ("User hasn't
    // played challenge yet", constaté le 22/10/2026).
    try {
      if (estRecapDefiDuJour(racineOnglets)) synchroniserLignesAmis(carte, equipe, actif);
    } catch (e) {
      console.warn("[LLGG] Ajout des coéquipiers impossible :", e);
    }
    const lignes = lignesClassement(carte, racineOnglets);

    let visibles = 0;
    lignes.forEach((ligne) => {
      const garder =
        !actif ||
        ligne.classList.contains("laligue-ligne-ami") || // ajoutée par nous, déjà filtrée sur l'équipe
        [...ligne.querySelectorAll(".laligue-badge")].some((b) => b.dataset.laligueEquipe === equipe.nom);
      ligne.style.display = garder ? "" : "none";
      // Récap du défi du jour : chaque joueur a une 2e ligne de détail
      // (manches, replay) juste après la sienne, sans pseudo dedans.
      const detail = ligne.nextElementSibling;
      if (detail && /expandedDetails__/.test(detail.className)) detail.style.display = garder ? "" : "none";
      if (garder) visibles++;
    });
    // Séparateurs ("…" ou "1003 hidden players" entre le top et sa propre
    // ligne) : sans objet ici.
    carte.querySelectorAll('[class*="dividerRow__"], [class*="_divider__"], [class*="hiddenPlayersRow__"]').forEach((sep) => {
      if (racineOnglets.contains(sep)) return;
      const cible = sep.closest("tr") || sep;
      cible.style.display = actif ? "none" : "";
    });

    // État vide explicite plutôt qu'une liste muette, au même niveau que
    // les lignes (une <tr> dans un tableau, un <div> sinon).
    const parentLignes = lignes[0]?.parentElement;
    let vide = carte.querySelector(".laligue-llgg-vide");
    if (actif && visibles === 0 && parentLignes) {
      if (!vide) {
        if (parentLignes.tagName === "TBODY") {
          vide = document.createElement("tr");
          const colonnes = parentLignes.closest("table")?.querySelectorAll("thead th").length || 4;
          vide.innerHTML = `<td colspan="${colonnes}">${escapeHtml(t("ongletLlggVide"))}</td>`;
        } else {
          vide = document.createElement("div");
          vide.textContent = t("ongletLlggVide");
        }
        vide.className = "laligue-llgg-vide";
        parentLignes.appendChild(vide);
      }
    } else if (vide) {
      vide.remove();
    }

    if (actif) {
      // Recalé à chaque passage : la largeur des onglets peut changer
      // (redimensionnement, ouverture/fermeture de la carte).
      const notre = racineOnglets.querySelector(".laligue-onglet-llgg");
      if (notre) {
        racineOnglets.style.setProperty("--llgg-tab-x", `${notre.offsetLeft}px`);
        racineOnglets.style.setProperty("--llgg-tab-w", `${notre.offsetWidth}px`);
      }
    }
  }

  function retirerOngletsLlgg() {
    document.querySelectorAll(".laligue-onglet-llgg").forEach((o) => {
      const racine = o.closest('[class*="tabs_root__"]');
      o.remove();
      intentionLlgg = false;
      if (racine && racine.dataset.laligueModeLlgg === "1") {
        afficherModeLlgg(racine, false);
        appliquerFiltreOngletLlgg(racine, { nom: "" });
      }
    });
  }

  function tenterInjectionOngletLlgg() {
    // Barres d'onglets de classement : reconnues à leurs onglets natifs
    // "Friends" et "All" (ids en -tab-friends / -tab-all).
    const racines = [...document.querySelectorAll('[class*="tabs_root__"]')].filter(
      (r) => r.querySelector('[role="tab"][id$="-tab-all"]') && r.querySelector('[role="tab"][id$="-tab-friends"]')
    );
    if (!racines.length) {
      if (document.querySelector(".laligue-onglet-llgg")) retirerOngletsLlgg();
      return;
    }
    if (!estActif() || !estFiltreResultatsActif() || !donneesLigueEnMemoire) {
      retirerOngletsLlgg();
      return;
    }

    racines.forEach((racine) => {
      const liste = racine.querySelector('[role="tablist"]');
      if (!liste) return;

      const carte = trouverCarteClassement(racine); // null tant que la liste n'est pas chargée
      const equipe = trouverMonEquipe(donneesLigueEnMemoire);
      if (!equipe) return; // joueur non identifié ou hors Ligue : pas d'onglet

      let notre = liste.querySelector(".laligue-onglet-llgg");
      if (!notre) {
        const modele = liste.querySelector('[role="tab"]');
        if (!modele) return;
        notre = document.createElement("button");
        notre.type = "button";
        notre.setAttribute("role", "tab");
        notre.className = `${modele.className} laligue-onglet-llgg`;
        notre.title = t("ongletLlggTitre", equipe.nom);
        const etiquette = document.createElement("span");
        etiquette.className = "laligue-onglet-llgg-tag";
        etiquette.textContent = equipe.tag;
        etiquette.style.background = equipe.couleurFond;
        etiquette.style.color = couleurTexteSure(equipe.couleurFond, equipe.couleurTexte);
        const enveloppe = document.createElement("span");
        enveloppe.appendChild(etiquette);
        if (estRecapDefiDuJour(racine)) installerSuiviDateDefi();
        notre.appendChild(enveloppe);
        // Activation dès le pointerdown, et pas seulement au click : leur
        // barre d'onglets défilante capture le pointeur au pointerdown, ce
        // qui redirige le click vers la barre et l'empêchait d'atteindre
        // notre bouton (bug du 19/10/2026). Le click reste écouté pour le
        // clavier (Entrée/Espace) ; l'horodatage évite un double déclenchement.
        let derniereActivation = 0;
        const activer = (e) => {
          if (e.type === "pointerdown" && e.button !== 0) return;
          e.preventDefault();
          e.stopPropagation();
          if (Date.now() - derniereActivation < 400) return;
          derniereActivation = Date.now();
          definirModeLlgg(racine, true);
          basculerSurToutSiBesoin(liste, true);
          appliquerFiltreOngletLlgg(racine, equipe);
        };
        notre.addEventListener("pointerdown", activer);
        notre.addEventListener("click", activer);
        // Placé entre "Club" et "All" (à la fin si "All" est introuvable).
        const toutNatif = liste.querySelector('[role="tab"][id$="-tab-all"]');
        if (toutNatif) liste.insertBefore(notre, toutNatif);
        else liste.appendChild(notre);

        // Couleur "inactive" relevée UNE fois sur un onglet natif inactif,
        // pour atténuer l'onglet natif qui reste techniquement actif côté
        // React. Pas à chaque scan : getComputedStyle force un recalcul de
        // style, et cette couleur ne change pas.
        const inactifNatif = liste.querySelector('[role="tab"][data-state="inactive"]:not(.laligue-onglet-llgg)');
        if (inactifNatif) racine.style.setProperty("--llgg-tab-inactif", getComputedStyle(inactifNatif).color);

        // Un vrai choix d'onglet natif par l'utilisateur sort de notre
        // mode. Nos événements synthétiques (activerOngletNatif) ne sont
        // pas "trusted" : écartés, ils ne coupent pas le mode qu'on active.
        if (!liste.dataset.laligueEcoute) {
          liste.dataset.laligueEcoute = "1";
          const sortir = (e) => {
            if (!e.isTrusted) return;
            const onglet = e.target instanceof Element && e.target.closest('[role="tab"]');
            if (onglet && !onglet.classList.contains("laligue-onglet-llgg") && racine.dataset.laligueModeLlgg === "1") {
              definirModeLlgg(racine, false);
              appliquerFiltreOngletLlgg(racine, equipe);
            }
          };
          liste.addEventListener("pointerdown", sortir, true);
          liste.addEventListener("keydown", (e) => {
            if (["ArrowLeft", "ArrowRight", "Home", "End", "Enter", " "].includes(e.key)) sortir(e);
          }, true);
        }
      }

      // À chaque scan, depuis l'intention (et non depuis le DOM, que React
      // peut remonter) : carte reconstruite, nouveau round, etc.
      afficherModeLlgg(racine, intentionLlgg);
      if (intentionLlgg) {
        detecterGrapheDansTout(racine, liste);
        basculerSurToutSiBesoin(liste, false);
      }
      synchroniserIconeInfo(notre, racine, equipe);
      appliquerFiltreOngletLlgg(racine, equipe, carte);
    });
  }

  // ---------------------------------------------------------------
  // Récap du défi du jour : coéquipiers amis, même masqués
  // ---------------------------------------------------------------
  // Le défi du jour réunit des dizaines de milliers de joueurs : "All"
  // n'en contient qu'une poignée (le haut du classement et les voisins du
  // joueur), le reste est derrière "N hidden players". Parcourir tout le
  // classement coûterait des centaines de requêtes de ~600 Ko (100
  // joueurs max par page, détail de partie inclus) : écarté (19/10/2026).
  // À la place, UNE requête : le classement "Friends", que le serveur
  // filtre lui-même et renvoie en entier (comme "Club"). On y prend les
  // membres de l'équipe absents de la page, ajoutés comme lignes.
  // Limite assumée, expliquée par l'icône de l'onglet : seuls les
  // coéquipiers qui sont aussi amis GeoGuessr peuvent être retrouvés.
  // Le rang renvoyé est relatif à la liste d'amis (constaté sur "Club" :
  // 1..n). Le rang GLOBAL est retrouvé à part (voir chercherRangsGlobaux).
  const SVG_INFO = `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" xmlns="http://www.w3.org/2000/svg"><circle cx="12" cy="12" r="10"/><line x1="12" y1="11" x2="12" y2="17"/><line x1="12" y1="7" x2="12.01" y2="7"/></svg>`;

  // Le récap est la seule barre d'onglets avec "My game" (-tab-me).
  function estRecapDefiDuJour(racineOnglets) {
    return !!racineOnglets.querySelector('[role="tab"][id$="-tab-me"]');
  }

  // Date du défi AFFICHÉ : le récap permet de naviguer entre les jours
  // sans changer d'URL ni recharger la page (bug du 19/10/2026 : nos
  // lignes restaient celles du jour courant). On la lit dans les requêtes
  // que le jeu envoie lui-même (…/leaderboard/all?dateStr=…), via la
  // Resource Timing API : simple observation, aucune interception de
  // fetch. Nos propres requêtes sont ignorées (urlsDefiInternes), pour
  // qu'une réponse tardive ne ramène pas l'ancienne date.
  // Repli : jour UTC (les défis se terminent à 23:59:59Z, cf. "endTime").
  let dateDefiVue = null;
  let suiviDateDefiInstalle = false;
  const urlsDefiInternes = new Set();
  const RE_DATE_DEFI = /\/daily-challenges\/leaderboard\/[a-z]+\?(?:[^#]*&)?dateStr=(\d{4}-\d{2}-\d{2})/;

  function noterRequeteDefi(url) {
    if (urlsDefiInternes.has(url)) return;
    const m = RE_DATE_DEFI.exec(url);
    if (m && m[1] !== dateDefiVue) {
      dateDefiVue = m[1];
      // Revenir sur un jour redonne droit à un essai (un échec a pu venir
      // d'une date transitoire).
      datesRangsEchouees.delete(m[1]);
      planifierScan();
    }
  }

  function installerSuiviDateDefi() {
    if (suiviDateDefiInstalle) return;
    suiviDateDefiInstalle = true;
    try {
      performance.getEntriesByType("resource").forEach((e) => noterRequeteDefi(e.name));
      new PerformanceObserver((liste) => liste.getEntries().forEach((e) => noterRequeteDefi(e.name))).observe({
        type: "resource",
      });
    } catch (e) {
      // API absente : on reste sur le repli (jour UTC)
    }
  }

  function dateDefiDuJour() {
    return dateDefiVue || new Date().toISOString().slice(0, 10);
  }

  // fetch vers l'API du défi, en notant l'URL comme "interne".
  function fetchDefi(chemin) {
    const url = `${location.origin}${URL_API_DEFI}${chemin}`;
    urlsDefiInternes.add(url);
    return fetch(url, { credentials: "include" });
  }

  // Une requête par date et par chargement de page, seulement quand
  // l'onglet LLGG est activé. On ne garde que les champs affichés : la
  // réponse inclut le détail complet de chaque partie.
  const amisDefiParDate = new Map(); // date -> tableau d'entrées (null = en cours)
  const planifierScan = grouperParFrame(() => scannerEtInjecter());

  function chargerAmisDefi(date) {
    if (amisDefiParDate.has(date)) return;
    amisDefiParDate.set(date, null);
    fetchDefi(`/friends?dateStr=${encodeURIComponent(date)}`)
      .then((r) => {
        if (!r.ok) console.warn(`[LLGG] Classement des amis du défi du jour indisponible (HTTP ${r.status})`);
        return r.ok ? r.json() : null;
      })
      .then((json) => {
        const brut = Array.isArray(json) ? json : json?.entries;
        return (Array.isArray(brut) ? brut : []).map((e) => ({
          userId: String(e.userId || ""),
          nick: String(e.nick || ""),
          pinUrl: String(e.pinUrl || ""),
          pays: String(e.countryCode || "").toLowerCase(),
          serie: Number(e.currentStreak) || 0,
          deplacements: Number(e.totalStepsCount) || 0,
          score: Number(e.totalScore) || 0,
          medaille: String(e.medal || "").toLowerCase(),
          distance: Number(e.totalDistance) || 0,
          temps: Number(e.totalTime) || 0,
          partie: /^[A-Za-z0-9]{8,32}$/.test(e.game?.token || "") ? e.game.token : null,
          manches: (Array.isArray(e.game?.player?.guesses) ? e.game.player.guesses : []).map((g) => ({
            score: Number(g.roundScoreInPoints) || 0,
            distance: Number(g.distanceInMeters) || 0,
            temps: Number(g.time) || 0,
            deplacements: Number(g.stepsCount) || 0,
          })),
        }));
      })
      .catch(() => [])
      .then((entrees) => {
        amisDefiParDate.set(date, entrees);
        planifierScan();
      });
  }

  function scoreDeLigne(ligne) {
    const texte = ligne.querySelector('[class*="_score__"]')?.textContent || "";
    return Number(texte.replace(/\D/g, "")) || 0;
  }

  // Ligne construite en clonant une ligne native (même rendu, mêmes
  // colonnes), puis vidée de tout ce qui appartient à l'autre joueur.
  // Détail dépliable (manches, total, replay) : clone d'une ligne de
  // détail native, valeurs remplacées. Les unités ("pts", "steps",
  // "min", "sec") sont relevées dans ce clone pour rester dans la langue
  // du site. Ouverture/fermeture gérées par nous (classe laligue-ouvert),
  // leur composant n'ayant pas la main sur ces lignes.
  function formaterDistanceDefi(metres) {
    if (metres < 1000) return `${Math.round(metres)} m`;
    const km = metres / 1000;
    return `${km < 10 ? km.toFixed(1) : Math.round(km).toLocaleString("en-US")} km`;
  }

  function construireDetailAmi(modeleDetail, entree) {
    const detail = modeleDetail.cloneNode(true);
    detail.className = `${modeleDetail.className} laligue-ligne-ami-detail`;
    detail.removeAttribute("style");
    // Unités relevées dans les textes natifs ("5,000 pts", "40 steps",
    // "2 min, 48 sec") : on retire les chiffres et la ponctuation.
    const mots = (texte) => (texte || "").match(/[^\d\s,.\u202f\u00a0]+/g) || [];
    const motPts = mots(detail.querySelector('[class*="roundScore__"]')?.textContent)[0] || "pts";
    const metas = [...detail.querySelectorAll('[class*="roundMeta__"]')].map((m) => [...m.children].map((c) => c.textContent));
    const motPas = mots(metas[0]?.[2])[0] || "steps";
    const tempsComplet = metas.map((m) => mots(m[1])).find((w) => w.length >= 2) || ["min", "sec"];
    const [motMin, motSec] = tempsComplet;
    const formaterTemps = (sec) => {
      const m = Math.floor(sec / 60);
      const r = Math.round(sec % 60);
      if (!m) return `${r} ${motSec}`;
      return r ? `${m} ${motMin}, ${r} ${motSec}` : `${m} ${motMin}`;
    };
    const remplir = (bloc, score, distance, temps, pas) => {
      const scoreEl = bloc.querySelector('[class*="roundScore__"]');
      if (scoreEl) scoreEl.textContent = `${score.toLocaleString("en-US")} ${motPts}`;
      const meta = bloc.querySelector('[class*="roundMeta__"]');
      if (meta) {
        meta.textContent = "";
        [formaterDistanceDefi(distance), formaterTemps(temps), `${pas} ${motPas}`].forEach((texte) => {
          const span = document.createElement("span");
          span.textContent = texte;
          meta.appendChild(span);
        });
      }
    };
    const blocsManches = [...detail.querySelectorAll('[class*="roundItem__"]:not([class*="totalItem__"])')];
    blocsManches.forEach((bloc, i) => {
      const manche = entree.manches[i];
      if (!manche) {
        bloc.remove();
        return;
      }
      remplir(bloc, manche.score, manche.distance, manche.temps, manche.deplacements);
    });
    const total = detail.querySelector('[class*="totalItem__"]');
    if (total) {
      total.querySelector('[class*="copy-rounds-button"]')?.remove(); // bouton natif sans son action
      remplir(total, entree.score, entree.distance, entree.temps, entree.deplacements);
    }
    const replay = detail.querySelector('a[href*="/replay"]');
    if (replay) {
      if (entree.partie) replay.href = `/game/${entree.partie}/replay`;
      else replay.closest('[class*="replayItem__"]')?.remove();
    }
    return detail;
  }

  function basculerDetailAmi(ligne) {
    const detail = ligne.nextElementSibling;
    if (!detail || !detail.classList.contains("laligue-ligne-ami-detail")) return;
    const ouvert = !detail.classList.contains("laligue-ouvert");
    detail.classList.toggle("laligue-ouvert", ouvert);
    ligne.classList.toggle("laligue-ouvert", ouvert);
  }

  // Nom du pays dans la langue de l'extension (une instance par langue).
  const nomsPays = {};
  function nomPaysDepuisCode(code) {
    const majuscule = code.toUpperCase();
    try {
      nomsPays[LANGUE] = nomsPays[LANGUE] || new Intl.DisplayNames([LANGUE], { type: "region" });
      return nomsPays[LANGUE].of(majuscule) || majuscule;
    } catch (e) {
      return majuscule; // code infranational (ex. gb-sct) ou API absente
    }
  }

    ligne.querySelectorAll(".laligue-badge, .laligue-badge-signaler").forEach((el) => el.remove());
    ligne.querySelectorAll("[data-laligue-traite]").forEach((el) => {

        avatar.src = `https://www.geoguessr.com/images/resize:auto:48:48/gravity:ce/plain/${entree.pinUrl}`;
      } else {
        avatar.removeAttribute("src");
      }
      avatar.alt = entree.nick;
    }

    // résolu par ID (fiable) par traiterWrapper au prochain scan.
    const pseudo = ligne.querySelector('[class*="nick_nick__"]');

      lien.appendChild(pseudo);

    const deplacements = ligne.querySelector('[class*="_steps__"]');

    const scoreEl = ligne.querySelector('[class*="medalScore__"]') || ligne.querySelector('[class*="_score__"]');

      scoreEl.textContent = "";
      if (imgMedaille) {
        scoreEl.appendChild(imgMedaille.cloneNode(true));
        scoreEl.style.color = imgMedaille.parentElement.style.color;
      } else {
        scoreEl.style.color = "";
      }
      scoreEl.appendChild(document.createTextNode(entree.score.toLocaleString("en-US")));
    }

    if (entree.manches.length) {
      ligne.addEventListener("click", (e) => {
        if (e.target instanceof Element && e.target.closest("a")) return;
        basculerDetailAmi(ligne);
      });
    } else {
      const caret = ligne.querySelector('[class*="caretCell__"]');
      if (caret) caret.textContent = "";
    }

  // Rang global d'un coéquipier : le classement "All" est trié par score
  // décroissant, donc une recherche dichotomique sur le score suffit,
  // avec des requêtes d'UN seul joueur (count=1, ~6 Ko) : environ
  // log2(nombre de joueurs) ≈ 16 requêtes, puis une page de 20 pour
  // trouver le joueur exact parmi les ex-æquo (≈ 20 requêtes pour 50 000
  // joueurs, espacées de 80 ms). Les coéquipiers sont
  // traités du meilleur au moins bon, chacun repartant de la borne du
  // précédent et des scores déjà vus. Une requête à la fois, lancée
  // seulement à l'activation de l'onglet LLGG, résultat gardé 30 min.
  const CLE_RANGS_DEFI = "laligue_rangs_defi";
  const DUREE_RANGS_DEFI_MS = 30 * 60 * 1000;
  const URL_API_DEFI = "/api/v3/challenges/daily-challenges/leaderboard";
  let rechercheRangsEnCours = false;
  // Un échec (réseau, format inattendu) n'est PAS retenté à chaque scan :
  // une seule tentative par date et par chargement de page.
  const datesRangsEchouees = new Set();

  function lireRangsDefi(date) {
    const cache = lireJsonStockage(CLE_RANGS_DEFI);
    if (!cache || cache.date !== date || Date.now() - cache.horodatage > DUREE_RANGS_DEFI_MS) return {};
    return cache.rangs || {};
  }

  function ecrireRangDefi(date, userId, valeur) {
    const rangs = { ...lireRangsDefi(date), [userId]: valeur };
    // Horodatage conservé seulement si le cache courant est encore
    // valide : sinon chaque écriture repartirait déjà périmée.
    const ancien = lireJsonStockage(CLE_RANGS_DEFI);
    const valide = ancien && ancien.date === date && Date.now() - ancien.horodatage <= DUREE_RANGS_DEFI_MS;
    const horodatage = valide ? ancien.horodatage : Date.now();
    try {
      localStorage.setItem(CLE_RANGS_DEFI, JSON.stringify({ date, horodatage, rangs }));
    } catch (e) {
      // pas grave : on refera la recherche
    }
  }

  const pause = (ms) => new Promise((r) => setTimeout(r, ms));

  // Requêtes de la recherche de rang : espacées (pas de rafale vers
  // GeoGuessr), et un seul nouvel essai après une pause si le serveur
  // répond 429 (trop de requêtes).
  async function requeteDefi(chemin) {
    await pause(80);
    let reponse = await fetchDefi(chemin);
    if (reponse.status === 429) {
      await pause(2000);
      reponse = await fetchDefi(chemin);
    }
    if (!reponse.ok) throw new Error(`HTTP ${reponse.status} sur ${chemin}`);
    return reponse.json();
  }

  async function entreesDefi(date, depuis, nombre) {
    const json = await requeteDefi(`/all/entries?dateStr=${encodeURIComponent(date)}&from=${depuis}&count=${nombre}`);
    const brut = Array.isArray(json) ? json : json?.entries;
    return (Array.isArray(brut) ? brut : []).map((e) => ({
      rang: Number(e.rank),
      userId: String(e.userId || ""),
      score: Number(e.totalScore) || 0,
    }));
  }

  async function chercherRangsGlobaux(date, cibles) {
    const connus = lireRangsDefi(date);
    const restantes = cibles.filter((c) => !connus[c.userId]).sort((a, b) => b.score - a.score);
    if (!restantes.length || rechercheRangsEnCours || datesRangsEchouees.has(date)) return;
    rechercheRangsEnCours = true;
    try {
      // Pas besoin du nombre total de joueurs : au-delà de la fin du
      // classement, le score vaut -1, ce qui borne la recherche.
      const scores = new Map(); // rang -> score, partagé entre les recherches
      const scoreAuRang = async (rang) => {
        if (!scores.has(rang)) {
          const [e] = await entreesDefi(date, rang, 1);
          scores.set(rang, e ? e.score : -1);
        }
        return scores.get(rang);
      };
      // Classement vide pour cette date (typiquement : date pas encore
      // relevée dans les requêtes du jeu) : on n'écrit rien plutôt qu'un
      // faux "≈1" (vu en transition le 19/10/2026).
      if ((await scoreAuRang(1)) < 0) throw new Error(`classement vide pour ${date}`);
      let borne = 1;
      for (const cible of restantes) {
        // Premier rang dont le score est <= au sien : début des ex-æquo.
        // D'abord une borne haute par doublement (1000, 2000, 4000…),
        // puis dichotomie entre les deux.
        let bas = borne;
        let haut = Math.max(borne, 1000);
        while ((await scoreAuRang(haut)) > cible.score) {
          bas = haut + 1;
          haut *= 2;
        }
        while (bas < haut) {
          const milieu = Math.floor((bas + haut) / 2);
          if ((await scoreAuRang(milieu)) <= cible.score) haut = milieu;
          else bas = milieu + 1;
        }
        let rang = null;
        for (let page = 0, depuis = bas; page < 3; page++, depuis += 20) {
          const entrees = await entreesDefi(date, depuis, 20);
          entrees.forEach((e) => scores.set(e.rang, e.score));
          const trouve = entrees.find((e) => e.userId === cible.userId);
          if (trouve) {
            rang = trouve.rang;
            break;
          }
          if (!entrees.length || entrees[entrees.length - 1].score < cible.score) break;
        }
        ecrireRangDefi(date, cible.userId, rang ? { rang, approx: false } : { rang: bas, approx: true });
        borne = bas;
        planifierScan();
      }
    } catch (e) {
      datesRangsEchouees.add(date);
      console.warn("[LLGG] Recherche des rangs du défi du jour interrompue :", e);
    } finally {
      rechercheRangsEnCours = false;
    }
  }

  function afficherRangAmi(ligne, userId) {
    const rangEl = ligne.querySelector('[class*="rankBadge__"]');
    if (!rangEl) return;
    const date = dateDefiDuJour();
    const info = lireRangsDefi(date)[userId];
    const echec = !info && datesRangsEchouees.has(date);
    const texte = info ? `${info.approx ? "≈" : ""}${info.rang}` : echec ? "–" : "…";
    const titre = info ? (info.approx ? t("rangAmiApprox") : "") : t(echec ? "rangAmiIndispo" : "rangAmiInconnu");
    if (rangEl.textContent !== texte) rangEl.textContent = texte;
    if (rangEl.title !== titre) rangEl.title = titre;
  }

  function synchroniserLignesAmis(carte, equipe, actif) {
    const tbody = carte.querySelector('[class*="leaderboard-primitives_table__"] tbody');
    if (!tbody) return;
    const date = dateDefiDuJour();
    // Lignes d'un autre jour (navigation dans le récap) : retirées.
    const existantes = [...tbody.querySelectorAll(".laligue-ligne-ami")].filter((l) => {
      if (actif && l.dataset.laligueDate === date) return true;
      l.remove();
      return false;
    });
    // Détails orphelins (leur ligne vient d'être retirée, ou React a
    // inséré une ligne entre les deux) : retirés aussi.
    tbody.querySelectorAll(".laligue-ligne-ami-detail").forEach((d) => {
      if (!d.previousElementSibling?.classList.contains("laligue-ligne-ami")) d.remove();
    });
    if (!actif) return;
    const entrees = amisDefiParDate.get(date);
    if (entrees === undefined) chargerAmisDefi(date);
    if (!entrees) return; // en cours : on repassera au scan suivant

    const natives = [...tbody.querySelectorAll('tr[class*="summaryRow__"]:not(.laligue-ligne-ami)')];
    const modele = natives[0];
    if (!modele) return;
    // Déjà affichés par le jeu : comparés par pseudo (leurs lignes n'ont
    // pas de lien /user/), plus soi-même (affiché "You").
    const pseudosNatifs = new Set(natives.map((l) => l.querySelector('[class*="nick_nick__"]')?.textContent.trim()));
    const monId = localStorage.getItem(CLE_MON_ID);
    const membres = new Set((equipe.membres || []).map((m) => m.userId).filter(Boolean));
    const dejaAjoutes = new Set(existantes.map((l) => l.dataset.laligueAmi));

    const aAfficher = entrees
      .filter((e) => membres.has(e.userId) && e.userId !== monId && /^[0-9a-f]{24}$/.test(e.userId))
      .filter((e) => !pseudosNatifs.has(e.nick));
    chercherRangsGlobaux(date, aAfficher);
    existantes.forEach((l) => afficherRangAmi(l, l.dataset.laligueAmi));
    aAfficher
      .filter((e) => !dejaAjoutes.has(e.userId))
      .forEach((e) => {
        const ligne = construireLigneAmi(modele, e, tbody);
        // Placée avant la première ligne native de score inférieur.
        const suivante = natives.find((l) => scoreDeLigne(l) < e.score);
        tbody.insertBefore(ligne, suivante || null);
        const modeleDetail = tbody.querySelector('tr[class*="expandedDetails__"]:not(.laligue-ligne-ami-detail)');
        if (modeleDetail && e.manches.length) tbody.insertBefore(construireDetailAmi(modeleDetail, e), suivante || null);
      });
  }

  // Entre les rounds du défi du jour, "All" n'affiche pas une liste mais
  // un graphique de répartition des scores (constaté le 22/10/2026) : rien
  // à filtrer. Détecté quand "All" est actif, sans tableau de joueurs,
  // avec un dessin (svg/canvas) à la place, deux scans d'affilée à plus

  let cheminGrapheTout = null; // pathname de la partie où "All" est un graphique

  // Icône ⓘ : explique pourquoi des coéquipiers peuvent manquer (limite
  // de GeoGuessr, pas un bug) — récap du défi du jour (classement tronqué),
  // rounds du défi du jour (repli sur "Friends"), et rounds des challenges

  }
  let grapheVuDepuis = 0;

  function ongletCibleLlgg() {
    return cheminGrapheTout === location.pathname ? "friends" : "all";
  }

  function detecterGrapheDansTout(racine, liste) {
    if (ongletCibleLlgg() !== "all") return;
    const tout = liste.querySelector('[role="tab"][id$="-tab-all"][aria-selected="true"]');
    const toolbar = racine.closest('[class*="round-leaderboard_toolbar__"]');
    const carte = toolbar && toolbar.parentElement;
    const graphe =
      tout &&
      carte &&
      !carte.querySelector('[class*="round-leaderboard_table__"]') &&
      [...carte.querySelectorAll("svg, canvas")].some((el) => !toolbar.contains(el));
    if (!graphe) {
      grapheVuDepuis = 0;
      return;
    }
    if (!grapheVuDepuis) {
      grapheVuDepuis = Date.now();
      setTimeout(planifierScan, 700); // re-vérification sans attendre une mutation
      return;
    }
    if (Date.now() - grapheVuDepuis < 600) return;
    cheminGrapheTout = location.pathname;
    grapheVuDepuis = 0;
    delete liste.dataset.laligueToutTente;
    basculerSurToutSiBesoin(liste, true);
  }

  // Le filtre LLGG porte sur la liste "All" (ou "Friends", voir
  // ongletCibleLlgg). On y bascule à l'activation, puis une seule fois
  // par barre d'onglets (si React l'a remontée sur un autre onglet) —
  // jamais en boucle, pour ne pas lutter contre leur composant si la
  // bascule échoue : on filtre alors la liste affichée.
  function basculerSurToutSiBesoin(liste, forcer) {
    const cible = liste.querySelector(`[role="tab"][id$="-tab-${ongletCibleLlgg()}"]`);
    if (!cible || cible.getAttribute("aria-selected") === "true") return;
    if (!forcer && liste.dataset.laligueToutTente) return;
    liste.dataset.laligueToutTente = "1";
    activerOngletNatif(cible);
  }

  function scannerEtInjecter() {
    // Pas de ":not([data-laligue-traite])" ici, volontairement : le
    // filtrage se fait par comparaison d'identité (chemin rapide ci-dessous,
    // puis dans traiterWrapper), pour détecter les éléments DOM recyclés
    // avec un nouveau joueur plutôt que de les ignorer définitivement.
    // Deux structures différentes contiennent un pseudo de joueur sur le
    // site : "player-nick_wrapper" (partout ailleurs) et
    // "activity-card_groupActor__" (résumés d'activités groupées sur
    // /me/activities, ex: "Played Duels 2 times" — pas de <a href> dedans,
    // donc résolution par pseudo uniquement pour celles-ci).
    // Réglage lu UNE fois par scan (et non une fois par pseudo dans
    // traiterWrapper) : un gros classement en compte des centaines.
    if (estActif()) {
      // Chemin rapide synchrone : un pseudo déjà traité pour le même joueur
      // est écarté ici, sans créer de promesse (traiterWrapper est async,
      // et un grand classement compte des centaines de pseudos par scan).
      // Même règle d'identité que dans traiterWrapper (userId || pseudo).
      document
        .querySelectorAll('[class*="player-nick_wrapper"], [class*="activity-card_groupActor__"]')
        .forEach((wrapper) => {
          if (wrapper.dataset.laligueTraite) {
            const identite = extraireUserId(wrapper) || wrapper.querySelector('[class*="nick_nick__"]')?.textContent.trim();
            if (identite && wrapper.dataset.laligueIdentite === identite) return;
          }
          traiterWrapper(wrapper);
        });
    }

    // Appelée inconditionnellement (pas seulement sur /user/ ou
    // /me/profile) : la fonction doit aussi tourner quand on QUITTE une
    // page de profil, pour retirer la carte flottante — sinon elle
    // resterait affichée sur n'importe quel autre écran, y compris en
    // pleine partie.
    tenterInjectionCarteProfilEntete();

    if (location.pathname.startsWith("/results/")) {
      tenterInjectionFiltreResultats();
      tenterInjectionSwitchLlgg();
    }

    // Classements entre les rounds / en fin de partie : pas de route
    // dédiée, on se fie à la présence de leur barre d'outils.
    tenterInjectionOngletLlgg();

    // Tuile flottante indépendante — affichée partout par défaut (pas de
    // dépendance à une page précise). Se masque elle-même si l'utilisateur
    // l'a fermée aujourd'hui, ou s'il n'y a plus rien de programmé.
    tenterInjectionTuileAccueil();

    tenterInjectionMenuHeader();
  }

  // Sélection de division sur le site de l'organisation
  // ---------------------------------------------------------------
  // Le script tourne aussi sur laliguegeoguessr.fr (voir @match) dans UN
  // SEUL but : quand on arrive depuis un de nos liens d'en-tête de
  // division, sélectionner la bonne division et défiler jusqu'à elle —
  // leur site ne le permet pas via l'URL (changement de division en JS
  // pur, et équipes dans un shadow DOM). Absolument rien d'autre de
  // notre extension ne s'exécute là-bas : pas de badge, pas de tuile,
  // pas d'observer, pas d'appel à l'API.
  //
  // ⚠️ Ce bloc dépend de la structure INTERNE de leur site, qui peut
  // changer sans préavis. Tout y est donc en échec silencieux : si un
  // sélecteur ne correspond plus, on ne fait rien du tout et la page
  // reste parfaitement utilisable telle quelle.

  function attendreElement(trouver, surTrouve, essaisMax = 40) {
    let essais = 0;
    (function reessayer() {
      let cible = null;
      try {
        cible = trouver();
      } catch (e) {
        return; // structure inattendue : on abandonne silencieusement
      }
      if (cible) {
        surTrouve(cible);
        return;
      }
      if (++essais >= essaisMax) return; // ~4s max, puis on laisse tomber
      setTimeout(reessayer, 100);
    })();
  }

  function demarrerSiteLigue() {
    const division = new URLSearchParams(location.search).get(PARAM_DIVISION);
    if (!division || !/^[1-5]$/.test(division)) return; // arrivée normale sur leur site : on ne touche à rien

    const chemin = location.pathname;

    if (chemin.includes("results.html")) {
      // Onglets de division en DOM classique, changement via leur propre
      // handler onclick — on simule le clic plutôt que de reproduire
      // leur logique interne.
      //
      // ⚠️ On attend que le chargement INITIAL (division 1, lancé tout
      // seul par leur composant) soit terminé avant de cliquer. Leur
      // loadDivision() n'annule pas un chargement en cours : cliquer trop
      // tôt lance deux chargements en parallèle, et le dernier à finir
      // écrase l'affichage. Les divisions 2 et 4 font une requête de
      // moins (currentDay = 1) et finissaient AVANT la division 1, qui
      // les écrasait — bug remonté le 19/10/2026. Signal de fin fiable :
      // leur .ranking n'est inséré dans le shadow root qu'à la toute fin
      // de loadDivision (replaceChildren).
      attendreElement(
        () => {
          const conteneur = document.querySelector("rankings-container");
          const racine = conteneur && conteneur.shadowRoot;
          if (!racine || !racine.querySelector(".ranking")) return null;
          return document.querySelector(`.divisions [data-division="${division}"]`);
        },
        (onglet) => {
          if (!onglet.classList.contains("active")) onglet.click();
          onglet.scrollIntoView({ behavior: "smooth", block: "center" });
        },
        150 // ~15 s : la division 1 enchaîne 5 requêtes séquentielles, 4 s ne suffisaient pas toujours
      );
      return;
    }

    if (chemin.includes("presentation.html")) {
      // Ici, pas d'onglets : toutes les divisions sont empilées à
      // l'intérieur du shadow DOM de <teams-container>. Il est déclaré
      // "open", donc son contenu reste accessible en JS — on cherche le
      // titre correspondant et on défile dessus.
      attendreElement(
        () => {
          const conteneur = document.querySelector("teams-container");
          const racine = conteneur && conteneur.shadowRoot;
          if (!racine) return null;
          return [...racine.querySelectorAll("h2")].find(
            (h2) => h2.textContent.trim().toLowerCase() === `division ${division}`
          );
        },
        (titre) => titre.scrollIntoView({ behavior: "smooth", block: "start" })
      );
    }
  }

  // ---------------------------------------------------------------
  // Observer global + démarrage
  // ---------------------------------------------------------------
  function demarrer() {
    ajouterStylesFondation();
    // Injecté ICI, inconditionnellement, plutôt que d'attendre le
    // premier badge créé (ajouterStylesBadge est aussi appelée plus
    // tard, mais son garde-fou par ID évite le doublon) — sinon la règle
    // CSS qui masque les tags n'existe pas encore si l'utilisateur
    // désactive l'affichage AVANT qu'un badge soit apparu quelque part,
    // et la désactivation reste alors sans effet visuel (bug confirmé
    // le 17/08/2026, remonté par des utilisateurs Firefox).
    ajouterStylesBadge();
    installerMasquageTuileSurEntete();
    appliquerEtatActif();
    scannerEtInjecter();
    const observer = new MutationObserver(grouperParFrame(scannerEtInjecter));
    observer.observe(document.body, { childList: true, subtree: true });

    // Précharge les données dès le démarrage plutôt que d'attendre le
    // premier wrapper détecté, pour que le tout premier badge affiché
    // n'attende pas un aller-retour réseau à vide. Un scan est relancé à
    // l'arrivée des données : les injections qui en dépendent de façon
    // synchrone (onglet/switch LLGG) ne seraient sinon retentées qu'à la
    // prochaine mutation DOM, qui peut ne jamais venir si la page est
    // déjà stable.
    obtenirDonneesLigue().then((donnees) => {
      if (donnees) scannerEtInjecter();
    });
  }

  // Aiguillage par site : sur laliguegeoguessr.fr, SEULE la sélection de
  // division s'exécute (voir demarrerSiteLigue) ; tout le reste de
  // l'extension est réservé à geoguessr.com.
  const pointDEntree = SUR_SITE_LIGUE ? demarrerSiteLigue : demarrer;

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", pointDEntree);
  } else {
    pointDEntree();
  }
})();
